# Script:      True Spherical Terrain Generator from TouchTerrain TIFF Data
# Version:     3.12
# Author:      Script by Jan Mestan, TerraMakers
# Description: Converts a TouchTerrain TIFF digital elevation model into a 3D
#              printable spherical terrain mesh. It handles reprojection,
#              true sphere scale calculation, triangulation, and prepares
#              the terrain for a closed cylindrical base.
# Part 1/2:    TIFF Reprojection, Spherical Transformation & Mesh Generation
# Part 2/2:    Cylindrical Base Integration & Mesh Closure

# ============================================================
# Terrain TIFF -> TRUE SPHERICAL TERRAIN
# PART 1/2
# ============================================================

!pip install rasterio pyproj numpy scipy trimesh pillow --quiet
import numpy as np
import rasterio
from rasterio.warp import calculate_default_transform, reproject, Resampling
import trimesh
from google.colab import files

# ============================================================
# USER PARAMETERS
# ============================================================

EARTH_RADIUS_KM = 6371.0
# CHANGE
MODEL_WIDTH_MM = 333.0
TOP_HEIGHT_MM = 50.0
# CHANGE
ZSCALE = 1.3

# ============================================================
# UPLOAD TIFF
# ============================================================

print("Upload TouchTerrain TIFF")
uploaded = files.upload()
tif_file = list(uploaded.keys())[0]
print("Loaded:", tif_file)

# ============================================================
# READ TIFF
# ============================================================

src = rasterio.open(tif_file)
print("\nOriginal CRS:")
print(src.crs)
elevation = src.read(1).astype(np.float64)
if src.nodata is not None:
    elevation[
        np.isclose(
            elevation,
            src.nodata
        )
    ] = np.nan
elevation[
    ~np.isfinite(elevation)
] = np.nan
print(
    "Elevation range:",
    np.nanmin(elevation),
    np.nanmax(elevation)
)

# ============================================================
# REPROJECT TO EPSG4326
# ============================================================

dst_crs="EPSG:4326"
transform, width, height = calculate_default_transform(
    src.crs,
    dst_crs,
    src.width,
    src.height,
    *src.bounds
)
reprojected=np.empty(
    (height,width),
    dtype=np.float64
)
reproject(
    source=elevation,
    destination=reprojected,
    src_transform=src.transform,
    src_crs=src.crs,
    dst_transform=transform,
    dst_crs=dst_crs,
    resampling=Resampling.bilinear
)
print(
    "Before upscale:",
    reprojected.shape
)

# ============================================================
# UPSCALE TERRAIN GRID - SMOOTHER CYLINDER CUT
# ============================================================

from scipy.ndimage import zoom
from affine import Affine
UPSCALE = 2
reprojected = zoom(
    reprojected,
    UPSCALE,
    order=3
)
transform = Affine(
    transform.a / UPSCALE,
    transform.b,
    transform.c,
    transform.d,
    transform.e / UPSCALE,
    transform.f
)
print(
    "After upscale:",
    reprojected.shape
)

# ============================================================
# CREATE LAT/LON GRID
# ============================================================

rows, cols = np.indices(
    reprojected.shape
)
xs, ys = rasterio.transform.xy(
    transform,
    rows,
    cols
)
lon=np.array(xs)
lat=np.array(ys)
lon=lon.reshape(
    reprojected.shape
)
lat=lat.reshape(
    reprojected.shape
)
mask=np.isfinite(
    reprojected
)
lon_valid = lon[mask]
lat_valid = lat[mask]
elev_valid = reprojected[mask]

# ============================================================
# SCALE CALCULATION
# ============================================================

lon_extent = (
    lon_valid.max()
    -
    lon_valid.min()
)
earth_circum_mm = (
    2*np.pi*
    EARTH_RADIUS_KM*
    1000000
)
from pyproj import Geod
geod = Geod(ellps="WGS84")
lon_min = lon_valid.min()
lon_max = lon_valid.max()
lat_center = np.mean(lat_valid)
_, _, distance_m = geod.inv(
    lon_min, lat_center,
    lon_max, lat_center
)
physical_width_mm = distance_m * 1000
scale = (
    MODEL_WIDTH_MM /
    physical_width_mm
)
MAP_SCALE = 1 / scale
print()
print("Model scale 1:")
print(1/scale)

# ============================================================
# CREATE SPHERICAL TERRAIN
# ============================================================

lon_r=np.deg2rad(
    lon_valid
)
lat_r=np.deg2rad(
    lat_valid
)
terrain_height_m = (
    elev_valid *
    ZSCALE
)
terrain_height_mm = (
    terrain_height_m *
    1000 /
    MAP_SCALE
)
earth_radius_mm = (
    EARTH_RADIUS_KM *
    1000000 /
    MAP_SCALE
)
radius = (
    earth_radius_mm +
    terrain_height_mm
)
X = (
    radius *
    np.cos(lat_r) *
    np.cos(lon_r)
)
Y = (
    radius *
    np.cos(lat_r) *
    np.sin(lon_r)
)
Z = (
    radius *
    np.sin(lat_r)
)
vertices=np.column_stack(
    [
        X,
        Y,
        Z
    ]
)

# ============================================================
# COMPUTE MODEL SPHERE RADIUS + TRUE SCALE
# ============================================================

distances = np.linalg.norm(vertices, axis=1)
model_radius_mm = np.mean(distances)
true_scale = model_radius_mm / (EARTH_RADIUS_KM * 1_000_000)
print()
print("=== TRUE SPHERE SCALE ===")
print("Model radius [mm]:", model_radius_mm)
print("Scale factor:", true_scale)
print("True map scale 1:", 1 / true_scale)

# ============================================================
# TRIANGULATION
# ============================================================

h,w = reprojected.shape
index=-np.ones(
    reprojected.shape,
    dtype=int
)
index[mask]=np.arange(
    np.sum(mask)
)
faces=[]
for i in range(h-1):
    for j in range(w-1):
        a=index[i,j]
        b=index[i,j+1]
        c=index[i+1,j]
        d=index[i+1,j+1]
        if (
            a>=0 and
            b>=0 and
            c>=0
        ):
            faces.append(
                [
                    a,
                    b,
                    c
                ]
            )
        if (
            b>=0 and
            c>=0 and
            d>=0
        ):
            faces.append(
                [
                    b,
                    d,
                    c
                ]
            )
faces=np.array(
    faces,
    dtype=np.int64
)
terrain=trimesh.Trimesh(
    vertices=vertices,
    faces=faces,
    process=False
)

# ============================================================
# CLEAN MESH - NEW TRIMESH API
# ============================================================

terrain.update_faces(
    terrain.unique_faces()
)
valid_faces = (
    terrain.area_faces > 1e-12
)
terrain.update_faces(
    valid_faces
)
terrain.remove_unreferenced_vertices()
print()
print("Terrain mesh:")
print(terrain)

# ============================================================
# ROTATE TERRAIN CENTER TO Z AXIS
# ============================================================

center = terrain.centroid.copy()
direction = (
    center /
    np.linalg.norm(center)
)
target=np.array(
    [
        0.0,
        0.0,
        1.0
    ]
)
axis=np.cross(
    direction,
    target
)
dot=np.clip(
    np.dot(direction,target),
    -1.0,
    1.0
)
angle=np.arccos(dot)
if np.linalg.norm(axis)>1e-12:
    axis/=np.linalg.norm(axis)
    matrix = trimesh.transformations.rotation_matrix(
        angle,
        axis
    )
    terrain.apply_transform(
        matrix
    )
# center origin
terrain.apply_translation(
    -terrain.centroid
)

# ============================================================
# MOVE HIGHEST POINT TO TOP_HEIGHT_MM mm
# ============================================================

max_z = terrain.vertices[:,2].max()
terrain.vertices[:,2] -= (
    max_z -
    TOP_HEIGHT_MM
)
print()
print(
    "Final terrain Z:",
    terrain.vertices[:,2].min(),
    terrain.vertices[:,2].max()
)
print()
print("PART 1/2 COMPLETE")
print("Variable ready:")
print("terrain")

# ============================================================
# TRUE CYLINDER CUT + CLOSED BASE (WATERTIGHT VERSION)
# ============================================================

import trimesh.repair as repair
import trimesh
import numpy as np
from scipy.interpolate import griddata
from scipy.spatial import Delaunay
from google.colab import files
import trimesh.repair as repair
from scipy.spatial import cKDTree
shell = terrain.copy()

# ------------------------------------------------------------
# PARAMETERS
# ------------------------------------------------------------

BASE_DIAMETER_MM = 180
BASE_RADIUS_MM = BASE_DIAMETER_MM / 2
SEGMENTS = 256

# ------------------------------------------------------------
# CREATE CYLINDER BORDER
# ------------------------------------------------------------

theta = np.linspace(
    0,
    2*np.pi,
    SEGMENTS,
    endpoint=False
)
circle_xy = np.column_stack(
    [
        BASE_RADIUS_MM*np.cos(theta),
        BASE_RADIUS_MM*np.sin(theta)
    ]
)

# ------------------------------------------------------------
# INTERPOLATE HEIGHT ON BORDER
# ------------------------------------------------------------

xy = shell.vertices[:, :2]
z = shell.vertices[:, 2]
circle_z = griddata(
    xy,
    z,
    circle_xy,
    method="linear"
)
missing = np.isnan(circle_z)
if np.any(missing):
    circle_z[missing] = griddata(
        xy,
        z,
        circle_xy[missing],
        method="nearest"
    )
top_loop = np.column_stack(
    [
        circle_xy,
        circle_z
    ]
)

# ------------------------------------------------------------
# CREATE TERRAIN INSIDE CYLINDER
# ------------------------------------------------------------

r = np.linalg.norm(
    shell.vertices[:,:2],
    axis=1
)
inside = r <= BASE_RADIUS_MM
terrain_points = shell.vertices[inside]
# remove points that would duplicate border (lepší verze)
from scipy.spatial import cKDTree
points_xy = terrain_points[:,:2]
tree = cKDTree(circle_xy)
dist, _ = tree.query(points_xy, k=1)
keep = dist > 1e-6
terrain_points = terrain_points[keep]

# ------------------------------------------------------------
# BUILD TOP CAP
# ------------------------------------------------------------

cap_vertices = np.vstack(
    [
        terrain_points,
        top_loop
    ]
)
top_boundary_offset = len(terrain_points)
tri = Delaunay(
    cap_vertices[:,:2]
)
top_faces = tri.simplices
centers = np.mean(
    cap_vertices[top_faces][:,:,:2],
    axis=1
)
valid = (
    np.linalg.norm(
        centers,
        axis=1
    )
    <= BASE_RADIUS_MM
)
top_faces = top_faces[valid]

# ------------------------------------------------------------
# BOTTOM DISK
# ------------------------------------------------------------

bottom_loop = np.column_stack(
    [
        BASE_RADIUS_MM*np.cos(theta),
        BASE_RADIUS_MM*np.sin(theta),
        np.zeros(SEGMENTS)
    ]
)
bottom_center = np.array(
    [
        [0,0,0]
    ]
)
bottom_vertices = np.vstack(
    [
        bottom_loop,
        bottom_center
    ]
)
bottom_offset = len(cap_vertices)
bottom_faces=[]
center_id = bottom_offset + SEGMENTS
for i in range(SEGMENTS):
    n=(i+1)%SEGMENTS
    bottom_faces.append(
        [
            center_id,
            bottom_offset+n,
            bottom_offset+i
        ]
    )

# ------------------------------------------------------------
# MERGE ALL VERTICES
# ------------------------------------------------------------

vertices=np.vstack(
    [
        cap_vertices,
        bottom_vertices
    ]
)
faces=list(
    top_faces
)

# ------------------------------------------------------------
# SIDE WALL
# ------------------------------------------------------------

for i in range(SEGMENTS):
    n=(i+1)%SEGMENTS
    top_a = top_boundary_offset+i
    top_b = top_boundary_offset+n
    bottom_a = bottom_offset+i
    bottom_b = bottom_offset+n
    faces.append(
        [
            top_a,
            top_b,
            bottom_b
        ]
    )
    faces.append(
        [
            top_a,
            bottom_b,
            bottom_a
        ]
    )
faces.extend(
    np.array(bottom_faces)
)

# ------------------------------------------------------------
# CREATE FINAL MESH
# ------------------------------------------------------------

faces=np.array(
    faces,
    dtype=np.int64
)
final_mesh = trimesh.Trimesh(
    vertices=vertices,
    faces=faces,
    process=False
)

# ------------------------------------------------------------
# CLEAN + ADVANCED REPAIR
# ------------------------------------------------------------

print("Cleaning mesh...")
final_mesh.update_faces(final_mesh.unique_faces())
final_mesh.remove_unreferenced_vertices()
final_mesh.merge_vertices()
final_mesh.merge_vertices(digits_vertex=6)
import trimesh.repair as repair
repair.fill_holes(final_mesh)
final_mesh.fix_normals()
final_mesh.update_faces(final_mesh.unique_faces())
face_lengths = np.array([len(f) for f in final_mesh.faces])
degenerate = face_lengths < 3
if np.any(degenerate):
    final_mesh.update_faces(~degenerate)
print("\n=== AFTER REPAIR ===")
print("Watertight:", final_mesh.is_watertight)
print("Euler number:", final_mesh.euler_number)
edges = final_mesh.edges_sorted
unique_edges, counts = np.unique(edges, axis=0, return_counts=True)
print("Boundary edges:", np.sum(counts == 1))
print("Non-manifold edges:", np.sum(counts > 2))
print(final_mesh)

# ------------------------------------------------------------
# VALIDATION
# ------------------------------------------------------------

print()
print("==============================")
print("MESH VALIDATION")
print("==============================")
print(
    "Euler:",
    final_mesh.euler_number
)
print(
    "Watertight:",
    final_mesh.is_watertight
)
edges = final_mesh.edges_sorted
unique_edges, counts = np.unique(
    edges,
    axis=0,
    return_counts=True
)
print(
    "Boundary edges:",
    np.sum(counts==1)
)
print(
    "Non-manifold edges:",
    np.sum(counts>2)
)
print(final_mesh)

# ------------------------------------------------------------
# EXPORT
# ------------------------------------------------------------

output_file = (
    "terrain_closed_cylinder.stl"
)
final_mesh.export(
    output_file
)
files.download(
    output_file
)

from pyproj import Geod
import numpy as np
print("\nGenerating KML circle...")

# --------------------------------------------------
# CENTER (same data as terrain)
# --------------------------------------------------

lat0 = np.mean(lat_valid)
lon0 = np.mean(lon_valid)

# --------------------------------------------------
# RADIUS (from cylinder, exact match!)
# --------------------------------------------------

BASE_RADIUS_MM = 180 / 2  # same as above
radius_m = BASE_RADIUS_MM / scale / 1000.0
print(f"Center: {lat0:.6f}, {lon0:.6f}")
print(f"Radius: {radius_m:.2f} m")

# --------------------------------------------------
# GEODETIC CIRCLE
# --------------------------------------------------

geod = Geod(ellps="WGS84")
angles = np.linspace(0, 360, 360)
lons = []
lats = []
for az in angles:
    lon, lat, _ = geod.fwd(lon0, lat0, az, radius_m)
    lons.append(lon)
    lats.append(lat)
# close polygon
lons.append(lons[0])
lats.append(lats[0])

# --------------------------------------------------
# KML FORMAT
# --------------------------------------------------

coords = ""
for lon, lat in zip(lons, lats):
    coords += f"{lon:.6f},{lat:.6f},0 "
kml = f"""<kml xmlns="http://opengis.net">
<Placemark>
<Polygon>
<outerBoundaryIs>
<LinearRing>
<coordinates>
{coords}
</coordinates>
</LinearRing>
</outerBoundaryIs>
</Polygon>
</Placemark>
</kml>
"""

# --------------------------------------------------
# SAVE + DOWNLOAD
# --------------------------------------------------

kml_file = "terrain_circle.kml"
with open(kml_file, "w") as f:
    f.write(kml)
print("KML circle saved!")
from google.colab import files
files.download(kml_file)
