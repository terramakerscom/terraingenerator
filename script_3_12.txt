# OUTER ANNULUS ONLY: make a smoothed copy of the elevation raster.
# The original "reprojected" array remains untouched and is used by INNER.
def gaussian_filter_nan(data, sigma):
    if sigma <= 0:
        return data.copy()

    valid = np.isfinite(data)
    values = np.where(valid, data, 0.0)

    smooth_values = gaussian_filter(values, sigma=sigma, mode="nearest")
    smooth_weights = gaussian_filter(valid.astype(np.float64), sigma=sigma, mode="nearest")

    out = np.full_like(data, np.nan, dtype=np.float64)
    np.divide(
        smooth_values,
        smooth_weights,
        out=out,
        where=smooth_weights > 1e-12,
    )

    # Preserve the original NoData footprint.
    out[~valid] = np.nan
    return out

# ============================================================
# EXACT TANGENT-PLANE ORIENTATION: LOCAL EAST-NORTH-UP (ENU)
# ============================================================
# Goal for every final STL:
#   +X = EAST
#   +Y = NORTH
#   +Z = UP (away from Earth's center)
#
# IMPORTANT GEOMETRIC CONDITION:
# The circular INNER/ANNULUS cut is centered on XY=(0, 0).  The tangent plane
# through that center must therefore be the XY plane, and its normal (+/-Z)
# must pass EXACTLY through the center of the modeled Earth sphere.
#
# The previous version first oriented the raster using a spherical pixel
# centroid and then translated by the full 3-D mesh centroid.  The X/Y part of
# that translation moved the Earth center away from the Z axis.  Consequently
# the normal through the cut center no longer pointed exactly at the sphere
# center.
#
# Fix: use the actual terrain-surface centroid (the same physical center that
# the old code subsequently moved to the origin) to define the radial UP
# direction BEFORE any translation.  We then build an exact ENU frame at that
# direction.  After the rotation, this center lies on the Z axis, so only a
# Z translation is needed.  The Earth center therefore remains exactly on the
# Z axis below the cut center.


def exact_local_enu_frame_from_center(center_xyz, lon_deg_for_pole_fallback, pole_eps=1e-10):
    center_xyz = np.asarray(center_xyz, dtype=np.float64).reshape(3)
    center_norm = np.linalg.norm(center_xyz)
    if not np.isfinite(center_norm) or center_norm < 1e-12:
        raise RuntimeError("Could not determine a valid 3-D center for the terrain patch.")

    # Radial direction from Earth center to the chosen cut center.
    up = center_xyz / center_norm

    center_lat_rad = np.arcsin(np.clip(up[2], -1.0, 1.0))
    horizontal_norm = np.hypot(up[0], up[1])

    if horizontal_norm > pole_eps:
        center_lon_rad = np.arctan2(up[1], up[0])

        # EAST = increasing longitude.
        east = np.array([
            -np.sin(center_lon_rad),
             np.cos(center_lon_rad),
             0.0,
        ], dtype=np.float64)
        east /= np.linalg.norm(east)

        # NORTH = projection of Earth's north axis onto the tangent plane.
        earth_north_axis = np.array([0.0, 0.0, 1.0], dtype=np.float64)
        north = earth_north_axis - np.dot(earth_north_axis, up) * up
        north_norm = np.linalg.norm(north)
        if north_norm < 1e-12:
            raise RuntimeError("Numerical failure while constructing local NORTH axis.")
        north /= north_norm

        pole_fallback_used = False
    else:
        # Exact/near pole: geographic EAST/NORTH is singular.  Keep a
        # deterministic reference meridian derived from the raster longitudes.
        lon_rad = np.deg2rad(np.asarray(lon_deg_for_pole_fallback, dtype=np.float64))
        mean_sin_lon = np.mean(np.sin(lon_rad))
        mean_cos_lon = np.mean(np.cos(lon_rad))
        lon_resultant = np.hypot(mean_sin_lon, mean_cos_lon)

        if lon_resultant > 1e-12:
            center_lon_rad = np.arctan2(mean_sin_lon, mean_cos_lon)
        else:
            center_lon_rad = 0.0

        east = np.array([
            -np.sin(center_lon_rad),
             np.cos(center_lon_rad),
             0.0,
        ], dtype=np.float64)
        east /= np.linalg.norm(east)

        # Keep a right-handed frame: east x north = up.
        north = np.cross(up, east)
        north /= np.linalg.norm(north)
        pole_fallback_used = True

    # Final orthogonalization for numerical safety.
    east = east - np.dot(east, up) * up
    east /= np.linalg.norm(east)
    north = np.cross(up, east)
    north /= np.linalg.norm(north)

    # Preserve the true geographic NORTH sign away from the pole.
    if not pole_fallback_used:
        earth_north_axis = np.array([0.0, 0.0, 1.0], dtype=np.float64)
        north_reference = earth_north_axis - np.dot(earth_north_axis, up) * up
        if np.dot(north, north_reference) < 0:
            east = -east
            north = -north

    center_lon_deg = np.rad2deg(center_lon_rad)
    center_lat_deg = np.rad2deg(center_lat_rad)

    # Rows transform Earth-centered XYZ -> local ENU coordinates.
    rotation_3x3 = np.vstack([east, north, up])

    # Sanity checks.
    identity_error = np.max(np.abs(rotation_3x3 @ rotation_3x3.T - np.eye(3)))
    determinant = np.linalg.det(rotation_3x3)
    if identity_error > 1e-9 or determinant < 0.999999:
        raise RuntimeError(
            f"Invalid ENU rotation matrix: orthogonality error={identity_error}, "
            f"det={determinant}"
        )

    return (
        rotation_3x3,
        center_lon_deg,
        center_lat_deg,
        pole_fallback_used,
        center_norm,
    )

def interpolate_terrain_height(points_xy, z_shift=0.0):
    values = np.asarray(_terrain_linear(points_xy), dtype=np.float64).reshape(-1)
    missing = ~np.isfinite(values)
    if np.any(missing):
        _, nearest_idx = _terrain_nearest_tree.query(points_xy[missing], k=1)
        values[missing] = _base_z[nearest_idx]
    if z_shift != 0.0:
        values = values + z_shift
    return values


def interpolate_annulus_terrain_height(points_xy, z_shift=0.0):
    values = np.asarray(
        _annulus_terrain_linear(points_xy),
        dtype=np.float64,
    ).reshape(-1)
    missing = ~np.isfinite(values)
    if np.any(missing):
        _, nearest_idx = _annulus_terrain_nearest_tree.query(points_xy[missing], k=1)
        values[missing] = _annulus_base_z[nearest_idx]
    if z_shift != 0.0:
        values = values + z_shift
    return values


def interpolate_annulus2_terrain_height(points_xy, z_shift=0.0):
    values = np.asarray(
        _annulus2_terrain_linear(points_xy),
        dtype=np.float64,

    missing = ~np.isfinite(values)

        values[missing] = _annulus2_base_z[nearest_idx]
    if z_shift != 0.0:
        values = values + z_shift
    return values

# VECTORIZED LOOP CONNECTIVITY
# ============================================================
def connect_loop_faces(offset_a, offset_b, segments=SEGMENTS, reverse=False):
    """Return the same two triangles per loop segment as the original Python loops."""
    i = np.arange(segments, dtype=np.int64)
    n = (i + 1) % segments
    a0 = offset_a + i
    a1 = offset_a + n
    b0 = offset_b + i
    b1 = offset_b + n
    if reverse:
        return np.column_stack((
            np.column_stack((a0, b1, a1)),
            np.column_stack((a0, b0, b1)),
        )).reshape(-1, 3)
    return np.column_stack((
        np.column_stack((a0, a1, b1)),
        np.column_stack((a0, b1, b0)),
    )).reshape(-1, 3)


def bottom_fan_faces(center_id, bottom_offset, segments=SEGMENTS):
    i = np.arange(segments, dtype=np.int64)
    n = (i + 1) % segments
    return np.column_stack((
        np.full(segments, center_id, dtype=np.int64),
        bottom_offset + n,
        bottom_offset + i,
    ))

# OUTPUT 1: INNER CLOSED CYLINDER
# ============================================================

def build_inner_model(
    shell,
    hole_centers_xy,
    base_diameter_mm,
    variant_name="INNER",
    hole_diameter_mm=None,
):
    """Build one INNER solid with one or more self-supporting blind holes.

    hole_centers_xy is an iterable of (x, y) centers in final model coordinates:
      +X = EAST, -X = WEST, +Y = NORTH, -Y = SOUTH.

    base_diameter_mm selects which INNER base diameter this variant uses.
    """
    BASE_RADIUS_MM = float(base_diameter_mm) / 2.0
    BEVEL_MM = INNER_BEVEL_MM
    BEVEL_SEGMENTS = INNER_BEVEL_SEGMENTS
    TOP_BEVEL_MM = INNER_TOP_BEVEL_MM
    TOP_BEVEL_SEGMENTS = INNER_TOP_BEVEL_SEGMENTS
    HOLE_DEPTH_MM = INNER_BOTTOM_HOLE_DEPTH_MM

    hole_centers_xy = np.asarray(hole_centers_xy, dtype=np.float64).reshape(-1, 2)

    # Conventional blind holes may now use either one shared diameter or one
    # diameter per hole. This keeps the original behavior for all existing
    # callers while allowing INNER B to use 4 mm EAST/WEST holes and a 7 mm
    # NORTH hole.
    if hole_diameter_mm is None:
        hole_diameters_mm = np.full(
            len(hole_centers_xy), INNER_BOTTOM_HOLE_DIAMETER_MM, dtype=np.float64
        )
    else:
        diam = np.asarray(hole_diameter_mm, dtype=np.float64)
        if diam.ndim == 0:
            hole_diameters_mm = np.full(
                len(hole_centers_xy), float(diam), dtype=np.float64
            )
        else:
            hole_diameters_mm = diam.reshape(-1)
            if len(hole_diameters_mm) != len(hole_centers_xy):
                raise ValueError(
                    "hole_diameter_mm must be a scalar or have one value per hole"
                )

    hole_radii_mm = hole_diameters_mm / 2.0

    theta = np.linspace(0, 2 * np.pi, SEGMENTS, endpoint=False)
    ct, st = np.cos(theta), np.sin(theta)

    # XY geometry is unchanged by the output-specific vertical shift.
    z_shift = shell.vertices[:, 2].max() - _BASE_MAX_Z

    if TOP_BEVEL_MM < 0 or TOP_BEVEL_MM >= BASE_RADIUS_MM:
        raise ValueError("Invalid INNER_TOP_BEVEL_MM")
    if TOP_BEVEL_SEGMENTS < 1:
        raise ValueError("INNER_TOP_BEVEL_SEGMENTS must be >= 1")

    top_inner_radius = BASE_RADIUS_MM - TOP_BEVEL_MM
    top_inner_xy = np.column_stack([top_inner_radius * ct, top_inner_radius * st])

    def interpolate_border_height(points_xy):
        return interpolate_terrain_height(points_xy, z_shift=z_shift)

    top_inner_z = interpolate_border_height(top_inner_xy)
    top_loop = np.column_stack([top_inner_xy, top_inner_z])

    r = np.linalg.norm(shell.vertices[:, :2], axis=1)
    terrain_points = shell.vertices[r <= top_inner_radius]

    points_xy = terrain_points[:, :2]
    tree = cKDTree(top_inner_xy)
    dist, _ = tree.query(points_xy, k=1)
    terrain_points = terrain_points[dist > 1e-6]

    cap_vertices = np.vstack([terrain_points, top_loop])
    top_boundary_offset = len(terrain_points)

    tri = Delaunay(cap_vertices[:, :2])
    top_faces = tri.simplices
    centers = np.mean(cap_vertices[top_faces][:, :, :2], axis=1)
    top_faces = top_faces[np.linalg.norm(centers, axis=1) <= top_inner_radius]

    top_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2) * k / TOP_BEVEL_SEGMENTS
            ring_r = (
                BASE_RADIUS_MM - TOP_BEVEL_MM
                + TOP_BEVEL_MM * np.sin(phi)
            )
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            ring_z = terrain_ring_z - bevel_drop
            top_bevel_rings.append(np.column_stack([ring_xy, ring_z]))

    if BEVEL_MM <= 0:
        bevel_rings = []
    else:
        if BEVEL_MM >= BASE_RADIUS_MM:
            raise ValueError("INNER_BEVEL_MM too large")
        if BEVEL_SEGMENTS < 1:
            raise ValueError("INNER_BEVEL_SEGMENTS must be >= 1")
        bevel_rings = []
        for k in range(BEVEL_SEGMENTS):
            phi = (np.pi / 2) * k / BEVEL_SEGMENTS
            ring_r = BASE_RADIUS_MM - BEVEL_MM + BEVEL_MM * np.cos(phi)
            ring_z = BEVEL_MM - BEVEL_MM * np.sin(phi)
            bevel_rings.append(np.column_stack([
                ring_r * ct,
                ring_r * st,
                np.full(SEGMENTS, ring_z),
            ]))

    bottom_radius = BASE_RADIUS_MM - max(BEVEL_MM, 0)
    bottom_loop = np.column_stack([
        bottom_radius * ct,
        bottom_radius * st,
        np.zeros(SEGMENTS),
    ])

    # ------------------------------------------------------------
    # SELF-SUPPORTING BLIND HOLES FROM THE UNDERSIDE
    # Diameter = 4 mm by default.
    # Cylindrical wall: Z=0 ... Z=HOLE_DEPTH_MM.
    # Above that, a 45-degree conical roof closes the cavity to a point.
    # This removes the horizontal blind-hole ceiling, so no bridge is required.
    # ------------------------------------------------------------
    if np.any(hole_radii_mm <= 0):
        raise ValueError("Every hole diameter must be > 0")
    if HOLE_DEPTH_MM <= 0:
        raise ValueError("INNER_BOTTOM_HOLE_DEPTH_MM must be > 0")

    # Every complete opening must remain inside the flat bottom and different
    # openings must not touch or overlap one another.
    center_distances = np.linalg.norm(hole_centers_xy, axis=1)
    outer_extents = center_distances + hole_radii_mm
    if np.any(outer_extents >= bottom_radius):
        bad = int(np.argmax(outer_extents))
        raise ValueError(
            f"{variant_name}: bottom hole {bad + 1} intersects the lower bevel: "
            f"center distance + hole radius = {outer_extents[bad]:.3f} mm, "
            f"flat-bottom radius = {bottom_radius:.3f} mm"
        )

    if len(hole_centers_xy) > 1:
        delta = hole_centers_xy[:, None, :] - hole_centers_xy[None, :, :]
        pair_dist = np.linalg.norm(delta, axis=2)
        np.fill_diagonal(pair_dist, np.inf)
        min_pair = float(np.min(pair_dist))
        radius_sums = hole_radii_mm[:, None] + hole_radii_mm[None, :]
        overlap = pair_dist <= radius_sums
        np.fill_diagonal(overlap, False)
        if np.any(overlap):
            ij = np.argwhere(overlap)[0]
            i_bad, j_bad = int(ij[0]), int(ij[1])
            raise ValueError(
                f"{variant_name}: bottom holes overlap or touch; "
                f"spacing={pair_dist[i_bad, j_bad]:.3f} mm, "
                f"required>{radius_sums[i_bad, j_bad]:.3f} mm"
            )

    hole_geometries = []

    for hole_index, hole_center_xy in enumerate(hole_centers_xy):
        hole_radius_mm = float(hole_radii_mm[hole_index])
        hole_xy = hole_center_xy + np.column_stack([
            hole_radius_mm * ct,
            hole_radius_mm * st,
        ])
        hole_bottom_loop = np.column_stack([hole_xy, np.zeros(SEGMENTS)])
        hole_end_loop = np.column_stack([
            hole_xy,
            np.full(SEGMENTS, HOLE_DEPTH_MM),
        ])
        # 45-degree self-supporting conical roof: vertical rise equals hole radius.
        hole_roof_tip_z = HOLE_DEPTH_MM + hole_radius_mm
        hole_roof_tip = np.array([[
            hole_center_xy[0],
            hole_center_xy[1],
            hole_roof_tip_z,
        ]], dtype=np.float64)

        # Make sure the deepest cone tip cannot break through the terrain surface.
        check_xy = np.vstack([hole_xy, hole_center_xy.reshape(1, 2)])
        local_top_z = interpolate_border_height(check_xy)
        if hole_roof_tip_z >= float(np.min(local_top_z)):
            raise ValueError(
                f"{variant_name}: bottom hole {hole_index + 1} self-supporting roof "
                "is too deep and would break through the terrain surface."
            )

        hole_geometries.append({
            "center": hole_center_xy,
            "bottom_loop": hole_bottom_loop,
            "end_loop": hole_end_loop,
            "roof_tip": hole_roof_tip,
        })

    parts = [cap_vertices]

    top_bevel_offsets = []
    for ring in top_bevel_rings:
        top_bevel_offsets.append(sum(len(x) for x in parts))
        parts.append(ring)

    bevel_offsets = []
    for ring in bevel_rings:
        bevel_offsets.append(sum(len(x) for x in parts))
        parts.append(ring)

    bottom_offset = sum(len(x) for x in parts)
    parts.append(bottom_loop)

    # Append each hole's opening loop, cylindrical end loop, and cone tip.
    for hole in hole_geometries:
        hole["bottom_offset"] = sum(len(x) for x in parts)
        parts.append(hole["bottom_loop"])
        hole["end_offset"] = sum(len(x) for x in parts)
        parts.append(hole["end_loop"])
        hole["roof_tip_id"] = sum(len(x) for x in parts)
        parts.append(hole["roof_tip"])

    all_vertices = np.vstack(parts)
    face_parts = [np.asarray(top_faces, dtype=np.int64)]

    if top_bevel_offsets:
        previous_offset = top_boundary_offset
        for current_offset in top_bevel_offsets:
            face_parts.append(connect_loop_faces(previous_offset, current_offset))
            previous_offset = current_offset
        side_top_offset = top_bevel_offsets[-1]
    else:
        side_top_offset = top_boundary_offset

    side_bottom_offset = bevel_offsets[0] if bevel_offsets else bottom_offset
    face_parts.append(connect_loop_faces(side_top_offset, side_bottom_offset))

    for a_off, b_off in zip(bevel_offsets[:-1], bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off))

    if bevel_offsets:
        face_parts.append(connect_loop_faces(bevel_offsets[-1], bottom_offset))

    # ------------------------------------------------------------
    # TRIANGULATE THE FLAT BOTTOM WITH 1 OR MORE CIRCULAR HOLES
    # ------------------------------------------------------------
    # The old one-hole version could connect the outer loop directly to one
    # inner loop. With two independent holes that construction would overlap.
    # Shapely 2.1 constrained Delaunay triangulation respects the outer circle
    # and every circular hole boundary exactly, giving one clean planar face.
    outer_xy = bottom_loop[:, :2]
    hole_xy_rings = [h["bottom_loop"][:, :2] for h in hole_geometries]
    bottom_polygon = Polygon(outer_xy, holes=[ring for ring in hole_xy_rings])
    if (not bottom_polygon.is_valid) or bottom_polygon.is_empty:
        raise RuntimeError(f"{variant_name}: invalid bottom polygon with holes")

    bottom_triangles = shapely.constrained_delaunay_triangles(bottom_polygon)

    # Coordinate -> mesh vertex id. Constrained triangulation uses the supplied
    # boundary vertices; rounding only protects against tiny floating-point
    # representation differences when coordinates come back from GEOS.
    xy_to_vertex = {}
    for j, xy in enumerate(outer_xy):
        xy_to_vertex[(round(float(xy[0]), 10), round(float(xy[1]), 10))] = bottom_offset + j
    for hole in hole_geometries:
        off = hole["bottom_offset"]
        for j, xy in enumerate(hole["bottom_loop"][:, :2]):
            xy_to_vertex[(round(float(xy[0]), 10), round(float(xy[1]), 10))] = off + j

    bottom_faces = []
    for tri_geom in bottom_triangles.geoms:
        coords = np.asarray(tri_geom.exterior.coords, dtype=np.float64)[:3, :2]
        ids = []
        for xy in coords:
            key = (round(float(xy[0]), 10), round(float(xy[1]), 10))
            if key not in xy_to_vertex:
                raise RuntimeError(
                    f"{variant_name}: bottom triangulation introduced an unexpected vertex {xy}"
                )
            ids.append(xy_to_vertex[key])

        # Bottom outward normal must point toward -Z. A positive signed XY area
        # is +Z, so reverse such triangles.
        p0, p1, p2 = coords
        signed2 = (
            (p1[0] - p0[0]) * (p2[1] - p0[1])
            - (p1[1] - p0[1]) * (p2[0] - p0[0])
        )
        if signed2 > 0:
            ids = [ids[0], ids[2], ids[1]]
        bottom_faces.append(ids)

    face_parts.append(np.asarray(bottom_faces, dtype=np.int64))

    # Add the cylindrical wall + 45-degree conical self-supporting roof.
    for hole in hole_geometries:
        face_parts.append(
            connect_loop_faces(
                hole["bottom_offset"],
                hole["end_offset"],
                reverse=True,
            )
        )

        i = np.arange(SEGMENTS, dtype=np.int64)
        n = (i + 1) % SEGMENTS
        face_parts.append(np.column_stack([
            hole["end_offset"] + i,
            np.full(SEGMENTS, hole["roof_tip_id"], dtype=np.int64),
            hole["end_offset"] + n,
        ]))

    all_faces = np.vstack(face_parts)

    mesh = trimesh.Trimesh(
        vertices=all_vertices,
        faces=all_faces,
        process=False,
    )

    print(f"Cleaning {variant_name} mesh...")
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    mesh.merge_vertices()
    mesh.merge_vertices(digits_vertex=6)
    repair.fill_holes(mesh)
    mesh.fix_normals()
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    return mesh

def _signed_area_xy(loop_xy):
    x = loop_xy[:, 0]
    y = loop_xy[:, 1]
    return 0.5 * np.sum(x * np.roll(y, -1) - np.roll(x, -1) * y)


def _make_curved_tube_loop(path_radius, start_angle, end_angle, widths,
                           entry_cap_radius, stop_cap_radius, cap_samples=25,
                            inner_widths=None):
    """Closed CCW footprint around a circular arc.

    start_angle is the LOCKED stop-pocket angle and end_angle is the ENTRY angle.
    With end_angle > start_angle, a fixed pin travels from entry toward the stop
    when the model itself is rotated counter-clockwise as viewed from the top.
    """
    widths = np.asarray(widths, dtype=np.float64).reshape(-1)
    if len(widths) < 3:
        raise ValueError("Keyhole path needs at least 3 side samples")
    angles = np.linspace(float(start_angle), float(end_angle), len(widths))
    normals = np.column_stack([np.cos(angles), np.sin(angles)])
    centers = float(path_radius) * normals

    outer_side = centers + widths[:, None] * normals
    if inner_widths is None:
        inner_widths = widths
    inner_widths = np.asarray(inner_widths, dtype=np.float64).reshape(-1)
    if len(inner_widths) != len(widths):
        raise ValueError("Inner and outer widths must have equal sample counts")
    inner_side = centers - inner_widths[:, None] * normals

    # ENTRY cap: outer -> inner, bulging in the forward tangent direction.
    pe = angles[-1]
    ne = np.array([np.cos(pe), np.sin(pe)], dtype=np.float64)
    te = np.array([-np.sin(pe), np.cos(pe)], dtype=np.float64)
    ce = centers[-1]
    aa = np.linspace(0.0, np.pi, int(cap_samples))
    entry_cap = np.array([
        ce + float(entry_cap_radius) * (np.cos(a) * ne + np.sin(a) * te)
        for a in aa[1:]
    ])

    # Return along inner side, excluding entry point already supplied by cap.
    inner_return = inner_side[-2::-1]

    # STOP cap: inner -> outer, bulging opposite the travel direction.
    ps = angles[0]
    ns = np.array([np.cos(ps), np.sin(ps)], dtype=np.float64)
    ts = np.array([-np.sin(ps), np.cos(ps)], dtype=np.float64)
    cs = centers[0]
    bb = np.linspace(0.0, np.pi, int(cap_samples))
    stop_cap = np.array([
        cs + float(stop_cap_radius) * (-np.cos(b) * ns - np.sin(b) * ts)
        for b in bb[1:-1]
    ])

    loop = np.vstack([outer_side, entry_cap, inner_return, stop_cap])
    if _signed_area_xy(loop) < 0:
        loop = loop[::-1].copy()
    return loop


def _triangulate_known_planar_polygon(outer_xy, holes_xy, outer_offset,
                                      hole_offsets, normal_up, variant_name):
    """Triangulate a planar polygon while reusing exact boundary vertex IDs."""
    holes_xy = list(holes_xy)
    poly = Polygon(outer_xy, holes=holes_xy)
    if poly.is_empty or not poly.is_valid:
        raise RuntimeError(f"{variant_name}: invalid planar polygon")

    mapping = {}
    for j, xy in enumerate(outer_xy):
        mapping[(round(float(xy[0]), 9), round(float(xy[1]), 9))] = outer_offset + j
    for ring, off in zip(holes_xy, hole_offsets):
        for j, xy in enumerate(ring):
            mapping[(round(float(xy[0]), 9), round(float(xy[1]), 9))] = off + j

    triangles = shapely.constrained_delaunay_triangles(poly)
    faces = []
    for tri_geom in triangles.geoms:
        c = np.asarray(tri_geom.exterior.coords, dtype=np.float64)[:3, :2]
        ids = []
        for xy in c:
            key = (round(float(xy[0]), 9), round(float(xy[1]), 9))
            if key not in mapping:
                raise RuntimeError(
                    f"{variant_name}: planar triangulation introduced unexpected vertex {xy}"
                )
            ids.append(mapping[key])
        signed2 = ((c[1,0]-c[0,0])*(c[2,1]-c[0,1])
                   - (c[1,1]-c[0,1])*(c[2,0]-c[0,0]))
        if (signed2 > 0) != bool(normal_up):
            ids = [ids[0], ids[2], ids[1]]
        faces.append(ids)
    return np.asarray(faces, dtype=np.int64)


def _connect_shelf_loops(head_offset, stem_offset, head_xy, stem_xy, vertices):
    """Connect matching loops with an upward-facing retaining shelf."""
    if len(head_xy) != len(stem_xy):
        raise RuntimeError("Keyhole shelf loops must have identical vertex counts")
    n = len(head_xy)
    faces = []
    for i in range(n):
        j = (i + 1) % n
        candidates = [
            [head_offset + i, head_offset + j, stem_offset + j],
            [head_offset + i, stem_offset + j, stem_offset + i],
        ]
        for ids in candidates:
            p = vertices[np.asarray(ids, dtype=np.int64)]
            normal_z = np.cross(p[1] - p[0], p[2] - p[0])[2]
            # Where the two loops coincide at the entry cap, degenerate shelf
            # triangles are intentionally omitted.
            if abs(normal_z) < 1e-12:
                continue
            if normal_z < 0:
                ids = [ids[0], ids[2], ids[1]]
            faces.append(ids)
    return np.asarray(faces, dtype=np.int64)


def build_inner_east_west_keyhole_model(
    shell,
    base_diameter_mm,
    variant_name="INNER EAST+WEST UNDERCUT KEYHOLES + NORTH HOLE",
    ew_path_radius_mm=None,
    north_hole_center_distance_mm=None,
    north_hole_diameter_mm=None,
):
    """Build undercut bayonet locks at EAST and WEST plus one classic NORTH hole.

    Bayonet locked stop positions:
      * EAST = angle   0 deg at the selected E/W path radius
      * WEST = angle 180 deg at the selected E/W path radius

    Every bayonet ENTRY is at a positive angular offset from its locked stop.
    Therefore, viewed from TOP, rotating the MODEL COUNTER-CLOCKWISE moves fixed
    pins clockwise relative to the model from each entry to its stop pocket.

    A separate conventional self-supporting blind hole is centered exactly
    NORTH at NORTH_EXTRA_HOLE_CENTER_DISTANCE_MM and has diameter
    NORTH_EXTRA_HOLE_DIAMETER_MM.
    """
    BASE_RADIUS_MM = float(base_diameter_mm) / 2.0
    BEVEL_MM = INNER_BEVEL_MM
    BEVEL_SEGMENTS = INNER_BEVEL_SEGMENTS
    TOP_BEVEL_MM = INNER_TOP_BEVEL_MM
    TOP_BEVEL_SEGMENTS = INNER_TOP_BEVEL_SEGMENTS

    ew_path_radius = float(
        EW_BAYONET_CENTER_DISTANCE_MM if ew_path_radius_mm is None else ew_path_radius_mm
    )
    head_radius = float(EW_BAYONET_HEAD_DIAMETER_MM) / 2.0
    stem_radius = float(EW_BAYONET_STEM_SLOT_WIDTH_MM) / 2.0
    lip = float(EW_BAYONET_LIP_THICKNESS_MM)
    head_depth = float(EW_BAYONET_HEAD_CLEARANCE_DEPTH_MM)
    rotation_deg = float(EW_BAYONET_ROTATION_DEG)

    if not (ew_path_radius > 0 and head_radius > 0 and stem_radius > 0):
        raise ValueError("Invalid bayonet dimensions")
    if stem_radius >= head_radius:
        raise ValueError("Stem slot must be narrower than the pin-head channel")
    if not (0 < lip < head_depth):
        raise ValueError("Need 0 < lip thickness < head clearance depth")
    if not (0 < rotation_deg < 90):
        raise ValueError("Bayonet rotation angle must be between 0 and 90 degrees")

    classic_hole_diameter = float(
        NORTH_EXTRA_HOLE_DIAMETER_MM if north_hole_diameter_mm is None else north_hole_diameter_mm
    )
    classic_hole_radius = classic_hole_diameter / 2.0
    classic_hole_distance = float(
        NORTH_EXTRA_HOLE_CENTER_DISTANCE_MM
        if north_hole_center_distance_mm is None
        else north_hole_center_distance_mm
    )
    classic_hole_depth = float(INNER_BOTTOM_HOLE_DEPTH_MM)
    if classic_hole_radius <= 0 or classic_hole_distance <= 0:
        raise ValueError("Invalid additional NORTH hole dimensions")

    theta = np.linspace(0, 2*np.pi, SEGMENTS, endpoint=False)
    ct, st = np.cos(theta), np.sin(theta)
    z_shift = shell.vertices[:, 2].max() - _BASE_MAX_Z

    def interpolate_border_height(points_xy):
        return interpolate_terrain_height(points_xy, z_shift=z_shift)

    # ---------- top terrain and outer body ----------
    top_inner_radius = BASE_RADIUS_MM - TOP_BEVEL_MM
    top_inner_xy = np.column_stack([top_inner_radius*ct, top_inner_radius*st])
    top_inner_z = interpolate_border_height(top_inner_xy)
    top_loop = np.column_stack([top_inner_xy, top_inner_z])

    r = np.linalg.norm(shell.vertices[:, :2], axis=1)
    terrain_points = shell.vertices[r <= top_inner_radius]
    if len(terrain_points):
        dist, _ = cKDTree(top_inner_xy).query(terrain_points[:, :2], k=1)
        terrain_points = terrain_points[dist > 1e-6]

    cap_vertices = np.vstack([terrain_points, top_loop])
    top_boundary_offset = len(terrain_points)
    tri = Delaunay(cap_vertices[:, :2])
    top_faces = tri.simplices
    centers = np.mean(cap_vertices[top_faces][:, :, :2], axis=1)
    top_faces = top_faces[np.linalg.norm(centers, axis=1) <= top_inner_radius]

    top_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi/2)*k/TOP_BEVEL_SEGMENTS
            ring_r = BASE_RADIUS_MM - TOP_BEVEL_MM + TOP_BEVEL_MM*np.sin(phi)
            ring_xy = np.column_stack([ring_r*ct, ring_r*st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM*(1.0-np.cos(phi))
            top_bevel_rings.append(np.column_stack([ring_xy, terrain_ring_z-bevel_drop]))

    bevel_rings = []
    if BEVEL_MM > 0:
        for k in range(BEVEL_SEGMENTS):
            phi = (np.pi/2)*k/BEVEL_SEGMENTS
            ring_r = BASE_RADIUS_MM - BEVEL_MM + BEVEL_MM*np.cos(phi)
            ring_z = BEVEL_MM - BEVEL_MM*np.sin(phi)
            bevel_rings.append(np.column_stack([
                ring_r*ct, ring_r*st, np.full(SEGMENTS, ring_z)
            ]))

    bottom_radius = BASE_RADIUS_MM - max(BEVEL_MM, 0.0)
    bottom_xy = np.column_stack([bottom_radius*ct, bottom_radius*st])
    bottom_loop = np.column_stack([bottom_xy, np.zeros(SEGMENTS)])

    # ---------- EAST + WEST bayonet footprints ----------
    rotation_rad = np.deg2rad(rotation_deg)
    # +X EAST=0°, +Y NORTH=90°, -X WEST=180°.
    # entry = stop + positive angle => model locks CCW viewed from TOP.
    lock_specs = [
        ("EAST", 0.0, ew_path_radius, int(EW_BAYONET_ARC_SAMPLES)),
        ("WEST", np.pi, ew_path_radius, int(EW_BAYONET_ARC_SAMPLES)),
    ]
    keyholes = []

    for label, stop_angle, path_radius, arc_samples in lock_specs:
        entry_angle = stop_angle + rotation_rad
        # Dense paired loops preserve the circular entry with sub-micron chord error.
        side_samples = max(256, arc_samples)
        cap_samples = 97

        head_widths = np.full(side_samples, head_radius, dtype=np.float64)
        head_xy = _make_curved_tube_loop(
            path_radius, stop_angle, entry_angle, head_widths,
            entry_cap_radius=head_radius,
            stop_cap_radius=head_radius,
            cap_samples=cap_samples,
        )

        # Bottom opening = narrow curved stem slot UNION full circular entry.
        # Intersect each radial ray with the entry circle analytically.  The
        # outer and inner widths differ slightly because the path is curved.
        # Keep paired vertices with the head loop for the retaining shelf.
        angles = np.linspace(stop_angle, entry_angle, side_samples)
        delta = entry_angle - angles
        projection = path_radius * np.cos(delta)
        discriminant = head_radius**2 - (path_radius * np.sin(delta))**2
        hit = discriminant >= 0.0
        half_chord = np.sqrt(np.maximum(discriminant, 0.0))
        outer_widths = np.where(hit, np.maximum(stem_radius,
            projection + half_chord - path_radius), stem_radius)
        inner_widths = np.where(hit, np.maximum(stem_radius,
            path_radius - projection + half_chord), stem_radius)
        # Exact common endpoints avoid tiny floating point shelf slivers.
        outer_widths[-1] = inner_widths[-1] = head_radius
        outer_widths[0] = inner_widths[0] = stem_radius
        opening_xy = _make_curved_tube_loop(
            path_radius, stop_angle, entry_angle, outer_widths,
            entry_cap_radius=head_radius,
            stop_cap_radius=stem_radius,
            cap_samples=cap_samples,
            inner_widths=inner_widths,
        )

        if len(head_xy) != len(opening_xy):
            raise RuntimeError(f"{variant_name} {label}: paired loops have different sizes")
        head_poly = Polygon(head_xy)
        opening_poly = Polygon(opening_xy)
        if not head_poly.is_valid or not opening_poly.is_valid:
            raise RuntimeError(f"{variant_name} {label}: invalid keyhole footprint")
        if not head_poly.covers(opening_poly):
            raise RuntimeError(f"{variant_name} {label}: stem opening is not inside head channel")

        entry_center = Point(path_radius * np.cos(entry_angle),
                             path_radius * np.sin(entry_angle))
        # A polygon approximates the circle; allow at most 0.01 mm radial error.
        if (not opening_poly.covers(entry_center) or
                opening_poly.boundary.distance(entry_center) < head_radius - 0.01):
            raise RuntimeError(f"{variant_name} {label}: circular entry clearance is too small")

        max_r = float(np.max(np.linalg.norm(head_xy, axis=1)))
        if max_r >= bottom_radius - 1e-6:
            raise ValueError(
                f"{variant_name} {label}: head channel reaches lower bevel: "
                f"max radius={max_r:.3f} mm, flat-bottom radius={bottom_radius:.3f} mm"
            )

        local_top = interpolate_border_height(head_xy)

        # Preserve the full hidden head-channel clearance up to head_depth, then
        # close the cavity with a 45-degree roof.  Because this footprint is a
        # curved tube, progressively reduce its half-width while keeping the
        # same centerline.  The final 0.2-mm-wide ridge is below one nozzle width.
        final_half_width = float(BAYONET_ROOF_FINAL_HALF_WIDTH_MM)
        if not (0.0 < final_half_width < head_radius):
            raise ValueError("BAYONET_ROOF_FINAL_HALF_WIDTH_MM must be > 0 and < head radius")
        roof_rise = head_radius - final_half_width
        roof_top_z = head_depth + roof_rise
        if roof_top_z >= float(np.min(local_top)):
            raise ValueError(
                f"{variant_name} {label}: self-supporting bayonet roof would break through terrain"
            )

        roof_steps = max(1, int(np.ceil(roof_rise / BAYONET_ROOF_STEP_MM)))
        roof_loops_xy = []
        for roof_k in range(1, roof_steps + 1):
            rise = min(roof_rise, roof_k * roof_rise / roof_steps)
            roof_half_width = head_radius - rise
            roof_widths = np.full(side_samples, roof_half_width, dtype=np.float64)
            roof_xy = _make_curved_tube_loop(
                path_radius, stop_angle, entry_angle, roof_widths,
                entry_cap_radius=roof_half_width,
                stop_cap_radius=roof_half_width,
                cap_samples=cap_samples,
            )
            roof_z = head_depth + rise
            roof_loops_xy.append((roof_xy, roof_z))

        keyholes.append({
            "label": label,
            "stop_angle": stop_angle,
            "entry_angle": entry_angle,
            "path_radius": path_radius,
            "head_xy": head_xy,
            "opening_xy": opening_xy,
            "head_poly": head_poly,
            "opening_poly": opening_poly,
            "roof_loops_xy": roof_loops_xy,
        })

    # Every hidden head channel must be independent.
    for i in range(len(keyholes)):
        for j in range(i+1, len(keyholes)):
            if keyholes[i]["head_poly"].intersects(keyholes[j]["head_poly"]):
                raise ValueError(
                    f"{variant_name}: {keyholes[i]['label']} and {keyholes[j]['label']} "
                    "hidden head channels overlap/touch"
                )

    # ---------- conventional NORTH Ø7 mm self-supporting blind hole ----------
    classic_center = np.array([0.0, classic_hole_distance], dtype=np.float64)
    classic_xy = classic_center + np.column_stack([
        classic_hole_radius*ct, classic_hole_radius*st
    ])
    classic_bottom_loop = np.column_stack([classic_xy, np.zeros(SEGMENTS)])
    classic_end_loop = np.column_stack([
        classic_xy, np.full(SEGMENTS, classic_hole_depth)
    ])
    classic_roof_tip_z = classic_hole_depth + classic_hole_radius
    classic_roof_tip = np.array([[
        classic_center[0], classic_center[1], classic_roof_tip_z
    ]], dtype=np.float64)

    if classic_hole_distance + classic_hole_radius >= bottom_radius - 1e-6:
        raise ValueError(
            f"{variant_name}: additional NORTH hole reaches lower bevel: "
            f"center+radius={classic_hole_distance + classic_hole_radius:.3f} mm, "
            f"flat-bottom radius={bottom_radius:.3f} mm"
        )
    local_top = interpolate_border_height(np.vstack([classic_xy, classic_center.reshape(1,2)]))
    if classic_roof_tip_z >= float(np.min(local_top)):
        raise ValueError(f"{variant_name}: additional NORTH self-supporting hole would break through terrain")

    classic_poly = Point(float(classic_center[0]), float(classic_center[1])).buffer(
        classic_hole_radius, quad_segs=max(16, SEGMENTS//4)
    )
    for kh in keyholes:
        if kh["head_poly"].intersects(classic_poly):
            raise ValueError(
                f"{variant_name}: additional NORTH hole overlaps/touches {kh['label']} bayonet head channel"
            )

    # ---------- body vertices ----------
    parts = [cap_vertices]
    face_parts = [np.asarray(top_faces, dtype=np.int64)]

    top_bevel_offsets = []
    for ring in top_bevel_rings:
        top_bevel_offsets.append(sum(len(x) for x in parts))
        parts.append(ring)
    bevel_offsets = []
    for ring in bevel_rings:
        bevel_offsets.append(sum(len(x) for x in parts))
        parts.append(ring)

    bottom_offset = sum(len(x) for x in parts)
    parts.append(bottom_loop)

    # Four main loops per bayonet, followed by self-supporting roof loops.
    for kh in keyholes:
        opening_xy = kh["opening_xy"]
        head_xy = kh["head_xy"]
        kh["opening_bottom_offset"] = sum(len(x) for x in parts)
        parts.append(np.column_stack([opening_xy, np.zeros(len(opening_xy))]))
        kh["opening_lip_offset"] = sum(len(x) for x in parts)
        parts.append(np.column_stack([opening_xy, np.full(len(opening_xy), lip)]))
        kh["head_lip_offset"] = sum(len(x) for x in parts)
        parts.append(np.column_stack([head_xy, np.full(len(head_xy), lip)]))
        kh["head_top_offset"] = sum(len(x) for x in parts)
        parts.append(np.column_stack([head_xy, np.full(len(head_xy), head_depth)]))
        kh["roof_offsets"] = []
        for roof_xy, roof_z in kh["roof_loops_xy"]:
            kh["roof_offsets"].append(sum(len(x) for x in parts))
            parts.append(np.column_stack([roof_xy, np.full(len(roof_xy), roof_z)]))

    # Classic hole opening loop, cylindrical end loop, and conical roof tip.
    classic_bottom_offset = sum(len(x) for x in parts)
    parts.append(classic_bottom_loop)
    classic_end_offset = sum(len(x) for x in parts)
    parts.append(classic_end_loop)
    classic_roof_tip_id = sum(len(x) for x in parts)
    parts.append(classic_roof_tip)

    # ---------- outer body faces ----------
    if top_bevel_offsets:
        prev = top_boundary_offset
        for cur in top_bevel_offsets:
            face_parts.append(connect_loop_faces(prev, cur))
            prev = cur
        side_top = top_bevel_offsets[-1]
    else:
        side_top = top_boundary_offset

    side_bottom = bevel_offsets[0] if bevel_offsets else bottom_offset
    face_parts.append(connect_loop_faces(side_top, side_bottom))
    for a_off, b_off in zip(bevel_offsets[:-1], bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off))
    if bevel_offsets:
        face_parts.append(connect_loop_faces(bevel_offsets[-1], bottom_offset))

    # Flat bottom contains two bayonet openings + one classic circular hole.
    face_parts.append(_triangulate_known_planar_polygon(
        bottom_xy,
        [kh["opening_xy"] for kh in keyholes] + [classic_xy],
        bottom_offset,
        [kh["opening_bottom_offset"] for kh in keyholes] + [classic_bottom_offset],
        normal_up=False,
        variant_name=variant_name,
    ))

    current_vertices = np.vstack(parts)
    for kh in keyholes:
        opening_xy = kh["opening_xy"]
        head_xy = kh["head_xy"]

        face_parts.append(connect_loop_faces(
            kh["opening_bottom_offset"], kh["opening_lip_offset"],
            segments=len(opening_xy), reverse=True,
        ))

        shelf_faces = _connect_shelf_loops(
            kh["head_lip_offset"], kh["opening_lip_offset"],
            head_xy, opening_xy, current_vertices,
        )
        if len(shelf_faces):
            face_parts.append(shelf_faces)

        face_parts.append(connect_loop_faces(
            kh["head_lip_offset"], kh["head_top_offset"],
            segments=len(head_xy), reverse=True,
        ))

        # 45-degree roof above the full-clearance head channel.
        previous_offset = kh["head_top_offset"]
        previous_xy = head_xy
        for roof_offset, (roof_xy, roof_z) in zip(kh["roof_offsets"], kh["roof_loops_xy"]):
            if len(previous_xy) != len(roof_xy):
                raise RuntimeError(f"{variant_name} {kh['label']}: bayonet roof loop size mismatch")
            face_parts.append(connect_loop_faces(
                previous_offset, roof_offset,
                segments=len(roof_xy), reverse=True,
            ))
            previous_offset = roof_offset
            previous_xy = roof_xy

        # Only a sub-nozzle-width ridge remains at the very top.
        face_parts.append(_triangulate_known_planar_polygon(
            previous_xy, [], previous_offset, [],
            normal_up=False, variant_name=f"{variant_name} {kh['label']} roof ridge",
        ))

    # Conventional NORTH hole cylindrical wall + 45-degree conical roof.
    face_parts.append(connect_loop_faces(
        classic_bottom_offset, classic_end_offset,
        segments=SEGMENTS, reverse=True,
    ))
    ii = np.arange(SEGMENTS, dtype=np.int64)
    nn = (ii + 1) % SEGMENTS
    face_parts.append(np.column_stack([
        classic_end_offset + ii,
        np.full(SEGMENTS, classic_roof_tip_id, dtype=np.int64),
        classic_end_offset + nn,
    ]))

    mesh = trimesh.Trimesh(
        vertices=np.vstack(parts),
        faces=np.vstack(face_parts),
        process=False,
    )
    print(f"Cleaning {variant_name} direct-mesh geometry...")
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    mesh.merge_vertices(digits_vertex=7)

    repair.fill_holes(mesh)
    mesh.fix_normals()
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()

    edges = mesh.edges_sorted
    _, counts = np.unique(edges, axis=0, return_counts=True)
    boundary_edges = int(np.sum(counts == 1))
    nonmanifold_edges = int(np.sum(counts > 2))
    if (not mesh.is_watertight) or boundary_edges or nonmanifold_edges:
        boundary_z = []
        if boundary_edges:
            boundary_edge_ids = np.unique(edges[np.asarray(counts) == 1])
            if len(boundary_edge_ids):
                bz = mesh.vertices[boundary_edge_ids, 2]
                boundary_z = [float(np.min(bz)), float(np.max(bz))]
        raise RuntimeError(
            f"{variant_name}: direct keyhole mesh is not watertight AFTER repair "
            f"(boundary={boundary_edges}, nonmanifold={nonmanifold_edges}, "
            f"boundary_z_range={boundary_z})"
        )
    return mesh

def build_annulus_model(shell):
    OUTER_RADIUS_MM = ANNULUS_OUTER_DIAMETER_MM / 2.0
    INNER_RADIUS_MM = ANNULUS_INNER_DIAMETER_MM / 2.0
    BEVEL_MM = ANNULUS_BEVEL_MM
    BEVEL_SEGMENTS = ANNULUS_BEVEL_SEGMENTS
    TOP_BEVEL_MM = ANNULUS_TOP_BEVEL_MM
    TOP_BEVEL_SEGMENTS = ANNULUS_TOP_BEVEL_SEGMENTS

    if INNER_RADIUS_MM <= 0:
        raise ValueError("ANNULUS_INNER_DIAMETER_MM must be > 0")
    if INNER_RADIUS_MM >= OUTER_RADIUS_MM:
        raise ValueError("Annulus inner diameter must be smaller than outer diameter")
    if BEVEL_MM < 0 or TOP_BEVEL_MM < 0:
        raise ValueError("Annulus bevels must be >= 0")
    if BEVEL_SEGMENTS < 1 or TOP_BEVEL_SEGMENTS < 1:
        raise ValueError("Annulus bevel segment counts must be >= 1")
    if INNER_RADIUS_MM + BEVEL_MM >= OUTER_RADIUS_MM - BEVEL_MM:
        raise ValueError("ANNULUS_BEVEL_MM is too large for annulus width")
    if INNER_RADIUS_MM + TOP_BEVEL_MM >= OUTER_RADIUS_MM - TOP_BEVEL_MM:
        raise ValueError("ANNULUS_TOP_BEVEL_MM is too large for annulus width")

    theta = np.linspace(0.0, 2.0 * np.pi, SEGMENTS, endpoint=False)
    ct, st = np.cos(theta), np.sin(theta)

    # OUTER ANNULUS uses only the annulus-smoothed terrain/interpolator.
    z_shift = shell.vertices[:, 2].max() - _ANNULUS_BASE_MAX_Z

    def interpolate_border_height(points_xy):
        return interpolate_annulus_terrain_height(points_xy, z_shift=z_shift)

    top_outer_radius = OUTER_RADIUS_MM - TOP_BEVEL_MM
    top_inner_radius = INNER_RADIUS_MM + TOP_BEVEL_MM

    top_outer_xy = np.column_stack([top_outer_radius * ct, top_outer_radius * st])
    top_inner_xy = np.column_stack([top_inner_radius * ct, top_inner_radius * st])
    top_outer_z = interpolate_border_height(top_outer_xy)
    top_inner_z = interpolate_border_height(top_inner_xy)
    top_outer_loop = np.column_stack([top_outer_xy, top_outer_z])
    top_inner_loop = np.column_stack([top_inner_xy, top_inner_z])

    r = np.linalg.norm(shell.vertices[:, :2], axis=1)
    terrain_points = shell.vertices[(r >= top_inner_radius) & (r <= top_outer_radius)]

    if len(terrain_points) > 0:
        points_xy = terrain_points[:, :2]
        outer_dist, _ = cKDTree(top_outer_xy).query(points_xy, k=1)
        inner_dist, _ = cKDTree(top_inner_xy).query(points_xy, k=1)
        terrain_points = terrain_points[(outer_dist > 1e-6) & (inner_dist > 1e-6)]

    cap_vertices = np.vstack([terrain_points, top_outer_loop, top_inner_loop])
    top_outer_boundary_offset = len(terrain_points)
    top_inner_boundary_offset = top_outer_boundary_offset + SEGMENTS

    tri = Delaunay(cap_vertices[:, :2])
    top_faces = tri.simplices
    centers = np.mean(cap_vertices[top_faces][:, :, :2], axis=1)
    center_r = np.linalg.norm(centers, axis=1)
    top_faces = top_faces[(center_r <= top_outer_radius) & (center_r >= top_inner_radius)]

    top_outer_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2.0) * k / TOP_BEVEL_SEGMENTS
            ring_r = OUTER_RADIUS_MM - TOP_BEVEL_MM + TOP_BEVEL_MM * np.sin(phi)
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            top_outer_bevel_rings.append(
                np.column_stack([ring_xy, terrain_ring_z - bevel_drop])
            )

    top_inner_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2.0) * k / TOP_BEVEL_SEGMENTS
            ring_r = INNER_RADIUS_MM + TOP_BEVEL_MM - TOP_BEVEL_MM * np.sin(phi)
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            top_inner_bevel_rings.append(
                np.column_stack([ring_xy, terrain_ring_z - bevel_drop])
            )

    bottom_outer_bevel_rings = []
    if BEVEL_MM > 0:
        for k in range(BEVEL_SEGMENTS):
            phi = (np.pi / 2.0) * k / BEVEL_SEGMENTS
            ring_r = OUTER_RADIUS_MM - BEVEL_MM + BEVEL_MM * np.cos(phi)
            ring_z = BEVEL_MM - BEVEL_MM * np.sin(phi)
            bottom_outer_bevel_rings.append(np.column_stack([
                ring_r * ct,
                ring_r * st,
                np.full(SEGMENTS, ring_z),
            ]))

    bottom_inner_bevel_rings = []
    if BEVEL_MM > 0:
        for k in range(BEVEL_SEGMENTS):
            phi = (np.pi / 2.0) * k / BEVEL_SEGMENTS
            ring_r = INNER_RADIUS_MM + BEVEL_MM - BEVEL_MM * np.cos(phi)
            ring_z = BEVEL_MM - BEVEL_MM * np.sin(phi)
            bottom_inner_bevel_rings.append(np.column_stack([
                ring_r * ct,
                ring_r * st,
                np.full(SEGMENTS, ring_z),
            ]))

    bottom_outer_radius = OUTER_RADIUS_MM - BEVEL_MM
    bottom_inner_radius = INNER_RADIUS_MM + BEVEL_MM
    bottom_outer_loop = np.column_stack([
        bottom_outer_radius * ct,
        bottom_outer_radius * st,
        np.zeros(SEGMENTS),
    ])
    bottom_inner_loop = np.column_stack([
        bottom_inner_radius * ct,
        bottom_inner_radius * st,
        np.zeros(SEGMENTS),
    ])

    parts = [cap_vertices]

    def append_rings(rings):
        offsets = []
        for ring in rings:
            offsets.append(sum(len(x) for x in parts))
            parts.append(ring)
        return offsets

    top_outer_bevel_offsets = append_rings(top_outer_bevel_rings)
    top_inner_bevel_offsets = append_rings(top_inner_bevel_rings)
    bottom_outer_bevel_offsets = append_rings(bottom_outer_bevel_rings)
    bottom_inner_bevel_offsets = append_rings(bottom_inner_bevel_rings)

    bottom_outer_offset = sum(len(x) for x in parts)
    parts.append(bottom_outer_loop)
    bottom_inner_offset = sum(len(x) for x in parts)
    parts.append(bottom_inner_loop)

    all_vertices = np.vstack(parts)
    face_parts = [np.asarray(top_faces, dtype=np.int64)]

    previous_offset = top_outer_boundary_offset
    for current_offset in top_outer_bevel_offsets:
        face_parts.append(connect_loop_faces(previous_offset, current_offset, reverse=False))
        previous_offset = current_offset
    outer_wall_top_offset = previous_offset

    previous_offset = top_inner_boundary_offset
    for current_offset in top_inner_bevel_offsets:
        face_parts.append(connect_loop_faces(previous_offset, current_offset, reverse=True))
        previous_offset = current_offset
    inner_wall_top_offset = previous_offset

    outer_wall_bottom_offset = (
        bottom_outer_bevel_offsets[0]
        if bottom_outer_bevel_offsets else bottom_outer_offset
    )
    inner_wall_bottom_offset = (
        bottom_inner_bevel_offsets[0]
        if bottom_inner_bevel_offsets else bottom_inner_offset
    )

    face_parts.append(connect_loop_faces(outer_wall_top_offset, outer_wall_bottom_offset, reverse=False))
    face_parts.append(connect_loop_faces(inner_wall_top_offset, inner_wall_bottom_offset, reverse=True))

    for a_off, b_off in zip(bottom_outer_bevel_offsets[:-1], bottom_outer_bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off, reverse=False))
    if bottom_outer_bevel_offsets:
        face_parts.append(connect_loop_faces(bottom_outer_bevel_offsets[-1], bottom_outer_offset, reverse=False))

    for a_off, b_off in zip(bottom_inner_bevel_offsets[:-1], bottom_inner_bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off, reverse=True))
    if bottom_inner_bevel_offsets:
        face_parts.append(connect_loop_faces(bottom_inner_bevel_offsets[-1], bottom_inner_offset, reverse=True))

    face_parts.append(connect_loop_faces(bottom_outer_offset, bottom_inner_offset, reverse=True))
    all_faces = np.vstack(face_parts)

    mesh = trimesh.Trimesh(
        vertices=all_vertices,
        faces=all_faces,
        process=False,
    )

    print("Cleaning annulus mesh...")
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    mesh.merge_vertices()
    mesh.merge_vertices(digits_vertex=6)
    repair.fill_holes(mesh)
    mesh.fix_normals()
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    return mesh

def build_annulus2_model(shell):
    OUTER_RADIUS_MM = ANNULUS2_OUTER_DIAMETER_MM / 2.0
    INNER_RADIUS_MM = ANNULUS2_INNER_DIAMETER_MM / 2.0
    BEVEL_MM = ANNULUS2_OUTER_BOTTOM_BEVEL_MM
    BEVEL_SEGMENTS = ANNULUS2_OUTER_BOTTOM_BEVEL_SEGMENTS
    TOP_BEVEL_MM = ANNULUS2_TOP_BEVEL_MM
    TOP_BEVEL_SEGMENTS = ANNULUS2_TOP_BEVEL_SEGMENTS

    if INNER_RADIUS_MM <= 0:
        raise ValueError("ANNULUS2_INNER_DIAMETER_MM must be > 0")
    if INNER_RADIUS_MM >= OUTER_RADIUS_MM:
        raise ValueError("Annulus 2 inner diameter must be smaller than outer diameter")
    if BEVEL_MM < 0 or TOP_BEVEL_MM < 0:
        raise ValueError("Annulus 2 bevels must be >= 0")
    if BEVEL_SEGMENTS < 1 or TOP_BEVEL_SEGMENTS < 1:
        raise ValueError("Annulus 2 bevel segment counts must be >= 1")
    if INNER_RADIUS_MM >= OUTER_RADIUS_MM - BEVEL_MM:
        raise ValueError("ANNULUS2_OUTER_BOTTOM_BEVEL_MM is too large for annulus width")
    if INNER_RADIUS_MM + TOP_BEVEL_MM >= OUTER_RADIUS_MM - TOP_BEVEL_MM:
        raise ValueError("ANNULUS2_TOP_BEVEL_MM is too large for annulus width")

    theta = np.linspace(0.0, 2.0 * np.pi, SEGMENTS, endpoint=False)
    ct, st = np.cos(theta), np.sin(theta)

    # OUTER ANNULUS uses only the annulus-smoothed terrain/interpolator.
    z_shift = shell.vertices[:, 2].max() - _ANNULUS2_BASE_MAX_Z

    def interpolate_border_height(points_xy):
        return interpolate_annulus2_terrain_height(points_xy, z_shift=z_shift)

    top_outer_radius = OUTER_RADIUS_MM - TOP_BEVEL_MM
    top_inner_radius = INNER_RADIUS_MM + TOP_BEVEL_MM

    top_outer_xy = np.column_stack([top_outer_radius * ct, top_outer_radius * st])
    top_inner_xy = np.column_stack([top_inner_radius * ct, top_inner_radius * st])
    top_outer_z = interpolate_border_height(top_outer_xy)
    top_inner_z = interpolate_border_height(top_inner_xy)
    top_outer_loop = np.column_stack([top_outer_xy, top_outer_z])
    top_inner_loop = np.column_stack([top_inner_xy, top_inner_z])

    r = np.linalg.norm(shell.vertices[:, :2], axis=1)
    terrain_points = shell.vertices[(r >= top_inner_radius) & (r <= top_outer_radius)]

    if len(terrain_points) > 0:
        points_xy = terrain_points[:, :2]
        outer_dist, _ = cKDTree(top_outer_xy).query(points_xy, k=1)
        inner_dist, _ = cKDTree(top_inner_xy).query(points_xy, k=1)
        terrain_points = terrain_points[(outer_dist > 1e-6) & (inner_dist > 1e-6)]

    cap_vertices = np.vstack([terrain_points, top_outer_loop, top_inner_loop])
    top_outer_boundary_offset = len(terrain_points)
    top_inner_boundary_offset = top_outer_boundary_offset + SEGMENTS

    tri = Delaunay(cap_vertices[:, :2])
    top_faces = tri.simplices
    centers = np.mean(cap_vertices[top_faces][:, :, :2], axis=1)
    center_r = np.linalg.norm(centers, axis=1)
    top_faces = top_faces[(center_r <= top_outer_radius) & (center_r >= top_inner_radius)]

    top_outer_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2.0) * k / TOP_BEVEL_SEGMENTS
            ring_r = OUTER_RADIUS_MM - TOP_BEVEL_MM + TOP_BEVEL_MM * np.sin(phi)
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            top_outer_bevel_rings.append(
                np.column_stack([ring_xy, terrain_ring_z - bevel_drop])
            )

    top_inner_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2.0) * k / TOP_BEVEL_SEGMENTS
            ring_r = INNER_RADIUS_MM + TOP_BEVEL_MM - TOP_BEVEL_MM * np.sin(phi)
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            top_inner_bevel_rings.append(
                np.column_stack([ring_xy, terrain_ring_z - bevel_drop])
            )

    bottom_outer_bevel_rings = []
    if BEVEL_MM > 0:
        for k in range(BEVEL_SEGMENTS):
            phi = (np.pi / 2.0) * k / BEVEL_SEGMENTS
            ring_r = OUTER_RADIUS_MM - BEVEL_MM + BEVEL_MM * np.cos(phi)
            ring_z = BEVEL_MM - BEVEL_MM * np.sin(phi)
            bottom_outer_bevel_rings.append(np.column_stack([
                ring_r * ct,
                ring_r * st,
                np.full(SEGMENTS, ring_z),
            ]))

    # ANNULUS 2: intentionally NO bottom inner bevel.
    # The inner cylindrical wall continues straight down to Z=0 at INNER_RADIUS_MM.
    bottom_inner_bevel_rings = []

    bottom_outer_radius = OUTER_RADIUS_MM - BEVEL_MM
    bottom_inner_radius = INNER_RADIUS_MM
    bottom_outer_loop = np.column_stack([
        bottom_outer_radius * ct,
        bottom_outer_radius * st,
        np.zeros(SEGMENTS),
    ])
    bottom_inner_loop = np.column_stack([
        bottom_inner_radius * ct,
        bottom_inner_radius * st,
        np.zeros(SEGMENTS),
    ])

    parts = [cap_vertices]

    def append_rings(rings):
        offsets = []
        for ring in rings:
            offsets.append(sum(len(x) for x in parts))
            parts.append(ring)
        return offsets

    top_outer_bevel_offsets = append_rings(top_outer_bevel_rings)
    top_inner_bevel_offsets = append_rings(top_inner_bevel_rings)
    bottom_outer_bevel_offsets = append_rings(bottom_outer_bevel_rings)
    bottom_inner_bevel_offsets = append_rings(bottom_inner_bevel_rings)

    bottom_outer_offset = sum(len(x) for x in parts)
    parts.append(bottom_outer_loop)
    bottom_inner_offset = sum(len(x) for x in parts)
    parts.append(bottom_inner_loop)

    all_vertices = np.vstack(parts)
    face_parts = [np.asarray(top_faces, dtype=np.int64)]

    previous_offset = top_outer_boundary_offset
    for current_offset in top_outer_bevel_offsets:
        face_parts.append(connect_loop_faces(previous_offset, current_offset, reverse=False))
        previous_offset = current_offset
    outer_wall_top_offset = previous_offset

    previous_offset = top_inner_boundary_offset
    for current_offset in top_inner_bevel_offsets:
        face_parts.append(connect_loop_faces(previous_offset, current_offset, reverse=True))
        previous_offset = current_offset
    inner_wall_top_offset = previous_offset

    outer_wall_bottom_offset = (
        bottom_outer_bevel_offsets[0]
        if bottom_outer_bevel_offsets else bottom_outer_offset
    )
    inner_wall_bottom_offset = (
        bottom_inner_bevel_offsets[0]
        if bottom_inner_bevel_offsets else bottom_inner_offset
    )

    face_parts.append(connect_loop_faces(outer_wall_top_offset, outer_wall_bottom_offset, reverse=False))
    face_parts.append(connect_loop_faces(inner_wall_top_offset, inner_wall_bottom_offset, reverse=True))

    for a_off, b_off in zip(bottom_outer_bevel_offsets[:-1], bottom_outer_bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off, reverse=False))
    if bottom_outer_bevel_offsets:
        face_parts.append(connect_loop_faces(bottom_outer_bevel_offsets[-1], bottom_outer_offset, reverse=False))

    for a_off, b_off in zip(bottom_inner_bevel_offsets[:-1], bottom_inner_bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off, reverse=True))
    if bottom_inner_bevel_offsets:
        face_parts.append(connect_loop_faces(bottom_inner_bevel_offsets[-1], bottom_inner_offset, reverse=True))

    face_parts.append(connect_loop_faces(bottom_outer_offset, bottom_inner_offset, reverse=True))
    all_faces = np.vstack(face_parts)

    mesh = trimesh.Trimesh(
        vertices=all_vertices,
        faces=all_faces,
        process=False,
    )

    print("Cleaning annulus 2 mesh...")
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    mesh.merge_vertices()
    mesh.merge_vertices(digits_vertex=6)
    repair.fill_holes(mesh)
    mesh.fix_normals()
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    return mesh

def _clean_boolean_mesh(mesh, title):
    """Light cleanup + strict topology check after a manifold boolean."""
    if isinstance(mesh, trimesh.Scene):
        mesh = trimesh.util.concatenate(tuple(mesh.geometry.values()))
    mesh.remove_unreferenced_vertices()
    mesh.merge_vertices(digits_vertex=7)
    repair.fill_holes(mesh)
    mesh.fix_normals()
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()

    edges = mesh.edges_sorted
    _, counts = np.unique(edges, axis=0, return_counts=True)
    boundary = int(np.sum(counts == 1))
    nonmanifold = int(np.sum(counts > 2))
    if (not mesh.is_watertight) or boundary or nonmanifold:
        raise RuntimeError(
            f"{title}: engraving boolean produced an invalid mesh "
            f"(watertight={mesh.is_watertight}, boundary={boundary}, nonmanifold={nonmanifold})"
        )
    return mesh


def _tm_to_manifold_direct(tm, label="mesh"):
    from manifold3d import Manifold as _M3DManifold, Mesh as _M3DMesh
    m3 = _M3DMesh(
        vert_properties=np.asarray(tm.vertices, dtype=np.float32),
        tri_verts=np.asarray(tm.faces, dtype=np.uint32),
    )
    solid = _M3DManifold(m3)
    if "NoError" not in str(solid.status()):
        raise RuntimeError(f"manifold3d rejected {label}: {solid.status()}")
    return solid


def _manifold_to_tm_direct(solid):
    raw = solid.to_mesh()
    return trimesh.Trimesh(
        vertices=np.asarray(raw.vert_properties, dtype=np.float64)[:, :3],
        faces=np.asarray(raw.tri_verts, dtype=np.int64),
        process=False,
    )


def _extrude_shapely_geom(geom, height, z0):
    pieces = []
    geoms = list(geom.geoms) if hasattr(geom, "geoms") else [geom]
    for poly in geoms:
        if poly.is_empty or poly.area <= 1e-10:
            continue
        if poly.geom_type != "Polygon":
            continue
        m = trimesh.creation.extrude_polygon(poly, height=float(height), engine="earcut")
        m.apply_translation([0.0, 0.0, float(z0)])
        pieces.append(m)
    if not pieces:
        return None
    return trimesh.util.concatenate(pieces)


def _make_self_supporting_recess_cutter(geom, label):
    """Straight 5 mm recess with vertical walls and a flat horizontal ceiling."""
    if geom is None or geom.is_empty:
        raise RuntimeError(f"Empty engraving footprint for {label}")
    if not geom.is_valid:
        geom = geom.buffer(0)
    if geom.is_empty:
        raise RuntimeError(f"Invalid engraving footprint for {label}")

    # Start slightly below Z=0 so the boolean cutter cleanly crosses the bottom face.
    # The top is EXACTLY at BOTTOM_ENGRAVING_DEPTH_MM, producing a flat ceiling.
    result = _extrude_shapely_geom(
        geom,
        BOTTOM_ENGRAVING_DEPTH_MM + BOTTOM_ENGRAVING_EPS_MM,
        -BOTTOM_ENGRAVING_EPS_MM,
    )
    if result is None:
        raise RuntimeError(f"Could not extrude engraving footprint for {label}")
    result.fix_normals(multibody=True)
    return result


def _planar_footprint_from_extruded_mesh(mesh, z_section_mm):
    """Recover the filled XY footprint (including counters/holes) from an extrusion."""
    section = mesh.section(
        plane_origin=[0.0, 0.0, float(z_section_mm)],
        plane_normal=[0.0, 0.0, 1.0],
    )
    if section is None:
        raise RuntimeError("Could not section engraving cutter for insert generation")

    geom = None
    for loop3d in section.discrete:
        loop_xy = np.asarray(loop3d, dtype=np.float64)[:, :2]
        if len(loop_xy) < 3:
            continue
        p = Polygon(loop_xy)
        if not p.is_valid:
            p = p.buffer(0)
        if p.is_empty or p.area <= 1e-10:
            continue
        geom = p if geom is None else geom.symmetric_difference(p)

    if geom is None or geom.is_empty:
        raise RuntimeError("Engraving cutter did not yield a usable insert footprint")
    if not geom.is_valid:
        geom = geom.buffer(0)
    return geom


def _make_bottom_insert_piece_from_cutter(cutter):
    """Create one 5 mm press-in artwork piece with the requested XY clearance."""
    if BOTTOM_INSERT_HEIGHT_MM <= 0:
        raise ValueError("BOTTOM_INSERT_HEIGHT_MM must be > 0")
    if BOTTOM_INSERT_TOTAL_XY_REDUCTION_MM < 0:
        raise ValueError("BOTTOM_INSERT_TOTAL_XY_REDUCTION_MM must be >= 0")

    geom = _planar_footprint_from_extruded_mesh(
        cutter,
        z_section_mm=0.5 * BOTTOM_ENGRAVING_DEPTH_MM,
    )

    # Half the requested total reduction is removed from every edge.
    # With 0.30 mm total reduction this gives 0.15 mm clearance per side.
    inset = 0.5 * float(BOTTOM_INSERT_TOTAL_XY_REDUCTION_MM)
    if inset > 0:
        geom = geom.buffer(-inset, join_style="mitre")
        if geom.is_empty:
            raise RuntimeError(
                "Insert clearance removed all artwork geometry; reduce "
                "BOTTOM_INSERT_TOTAL_XY_REDUCTION_MM."
            )
        if not geom.is_valid:
            geom = geom.buffer(0)

    piece = _extrude_shapely_geom(geom, BOTTOM_INSERT_HEIGHT_MM, 0.0)
    if piece is None:
        raise RuntimeError("Could not extrude separate bottom logo/text insert piece")
    piece.remove_unreferenced_vertices()
    piece.merge_vertices()
    piece.fix_normals(multibody=True)
    return piece


def _make_compact_bottom_insert_from_cutters(cutters):
    """Arrange press-in artwork compactly for printing.

    All normal rows use an exact 5 mm edge-to-edge vertical gap.  The final curved
    material-text row is handled specially: its CENTER section is placed 5 mm below
    the preceding N marker.  Using the global top of an arc would leave a large
    visual hole below N because the arc ends sit much higher than its middle.

    This affects ONLY the separate print-layout STL.  Engraving locations in both
    INNER models are unchanged.
    """
    gap = float(BOTTOM_INSERT_LAYOUT_GAP_MM)
    if gap < 0:
        raise ValueError("BOTTOM_INSERT_LAYOUT_GAP_MM must be >= 0")
    if not cutters:
        raise RuntimeError("No artwork cutters supplied for compact insert layout")

    pieces = []
    previous_bottom_y = None
    previous_piece = None

    for index, cutter in enumerate(cutters):
        piece = _make_bottom_insert_piece_from_cutter(cutter)

        # Center every complete artwork row on X=0.
        center_x = 0.5 * float(piece.bounds[0, 0] + piece.bounds[1, 0])
        piece.apply_translation([-center_x, 0.0, 0.0])

        is_last_piece = (index == len(cutters) - 1)
        is_curved_material = bool(BOTTOM_PLA_CURVED and is_last_piece)

        if previous_bottom_y is None:
            # Put the top edge of the logo/first item at Y=0 and stack downward.
            dy = -float(piece.bounds[1, 1])
        elif is_curved_material:
            # For the curved line, anchor the CENTRAL part of the arc under the
            # previous row (normally N).  This makes the visible central gap 5 mm
            # instead of measuring against the much higher arc endpoints.
            prev_width = float(previous_piece.bounds[1, 0] - previous_piece.bounds[0, 0])
            center_half_width = max(6.0, 0.75 * prev_width)
            verts = piece.vertices
            central = np.abs(verts[:, 0]) <= center_half_width
            if not np.any(central):
                # Robust fallback: use the 10% of vertices closest to X=0.
                order = np.argsort(np.abs(verts[:, 0]))
                count = max(1, int(np.ceil(0.10 * len(order))))
                central = np.zeros(len(verts), dtype=bool)
                central[order[:count]] = True
            central_top_y = float(np.max(verts[central, 1]))
            dy = (previous_bottom_y - gap) - central_top_y
        else:
            # Exact edge-to-edge gap between ordinary consecutive rows.
            dy = (previous_bottom_y - gap) - float(piece.bounds[1, 1])

        piece.apply_translation([0.0, dy, 0.0])
        previous_bottom_y = float(piece.bounds[0, 1])
        previous_piece = piece
        pieces.append(piece)

    insert = trimesh.util.concatenate(pieces)
    insert.remove_unreferenced_vertices()
    insert.merge_vertices()
    insert.fix_normals(multibody=True)
    print(
        f"Compact insert STL layout: {len(pieces)} artwork rows; "
        f"ordinary gaps={gap:.2f} mm and curved-text CENTER gap={gap:.2f} mm."
    )
    return insert


def _text_filled_geometry(text):
    """Convert TrueType text contours to a filled Shapely geometry."""
    prop = FontProperties(
        family=BOTTOM_TEXT_FONT_FAMILY,
        weight="bold" if BOTTOM_TEXT_BOLD else "normal",
    )
    path = TextPath((0.0, 0.0), str(text), size=1.0, prop=prop)
    contours = path.to_polygons()
    geom = None
    for contour in contours:
        if len(contour) < 3:
            continue
        p = Polygon(contour)
        if p.is_empty or p.area <= 1e-12:
            continue
        if not p.is_valid:
            p = p.buffer(0)
        if p.is_empty:
            continue
        geom = p if geom is None else geom.symmetric_difference(p)
    if geom is None or geom.is_empty:
        raise RuntimeError(f"Could not create vector geometry for text: {text!r}")
    if not geom.is_valid:
        geom = geom.buffer(0)
    return geom


def _scaled_text_geometry(text, target_height_mm=None, max_width_mm=None, xy_scale=1.0,
                          y_anchor="center"):
    geom = _text_filled_geometry(text)
    minx, miny, maxx, maxy = geom.bounds
    width = maxx - minx
    height = maxy - miny
    if width <= 0 or height <= 0:
        raise RuntimeError(f"Invalid text bounds for {text!r}: {geom.bounds}")
    if xy_scale <= 0:
        raise ValueError("Text xy_scale must be > 0")

    if target_height_mm is None:
        target_height_mm = BOTTOM_TEXT_HEIGHT_MM * xy_scale
    if max_width_mm is None:
        max_width_mm = BOTTOM_TEXT_MAX_WIDTH_MM * xy_scale

    factor = target_height_mm / height
    if width * factor > max_width_mm:
        factor = max_width_mm / width
    geom = shapely_scale(geom, xfact=factor, yfact=factor, origin=(0.0, 0.0))

    minx, miny, maxx, maxy = geom.bounds
    cx = 0.5 * (minx + maxx)
    if y_anchor == "center":
        yoff = -0.5 * (miny + maxy)
    elif y_anchor == "bottom":
        yoff = -miny
    else:
        raise ValueError(f"Unsupported y_anchor={y_anchor!r}")
    geom = shapely_translate(geom, xoff=-cx, yoff=yoff)
    return geom


def _make_text_cutter(text, center_y_mm, target_height_mm=None, max_width_mm=None,
                      xy_scale=1.0):
    geom = _scaled_text_geometry(
        text,
        target_height_mm=target_height_mm,
        max_width_mm=max_width_mm,
        xy_scale=xy_scale,
        y_anchor="center",
    )
    geom = shapely_translate(geom, xoff=0.0, yoff=center_y_mm)
    if BOTTOM_MIRROR_X_FOR_UNDERSIDE:
        geom = shapely_scale(geom, xfact=-1.0, yfact=1.0, origin=(0.0, 0.0))

    return _make_self_supporting_recess_cutter(geom, f"text {text!r}")


def _make_curved_text_cutter(text, outer_radius_mm, center_angle_deg=-90.0,
                             target_height_mm=None, max_width_mm=None, xy_scale=1.0):
    if outer_radius_mm <= 0:
        raise ValueError("outer_radius_mm must be > 0")

    geom = _scaled_text_geometry(
        text,
        target_height_mm=target_height_mm,
        max_width_mm=max_width_mm,
        xy_scale=xy_scale,
        y_anchor="bottom",
    )

    center_angle_rad = np.deg2rad(float(center_angle_deg))
    outer_radius_mm = float(outer_radius_mm)

    def _curve_coords(coords):
        arr = np.asarray(coords, dtype=np.float64).copy()
        x = arr[:, 0]
        y = arr[:, 1]
        theta = center_angle_rad + x / outer_radius_mm
        radius = outer_radius_mm - y
        arr[:, 0] = radius * np.cos(theta)
        arr[:, 1] = radius * np.sin(theta)
        return arr

    geom = shapely.transform(geom, _curve_coords)
    if BOTTOM_MIRROR_X_FOR_UNDERSIDE:
        geom = shapely_scale(geom, xfact=-1.0, yfact=1.0, origin=(0.0, 0.0))

    return _make_self_supporting_recess_cutter(geom, f"curved text {text!r}")

# Only the north marker "N" is engraved on both INNER undersides.
# The separate press-in STL contains exactly the same single "N".
_BOTTOM_ENGRAVING_CUTTER_CACHE = {}
_BOTTOM_ENGRAVING_INSERT_CACHE = {}

def engrave_inner_bottom(mesh, variant_name):
    if not BOTTOM_ENGRAVING_ENABLED:
        return mesh

    print(f"\nEngraving only N on the underside of {variant_name}...")
    cache_key = (
        str(BOTTOM_NORTH_MARKER_TEXT),
        float(BOTTOM_NORTH_MARKER_SCALE),
        float(BOTTOM_TEXT_SIZE_REFERENCE_WIDTH_MM),
        str(BOTTOM_TEXT_SIZE_REFERENCE),
        str(BOTTOM_TEXT_FONT_FAMILY),
        bool(BOTTOM_TEXT_BOLD),
        bool(BOTTOM_MIRROR_X_FOR_UNDERSIDE),
        float(BOTTOM_ENGRAVING_CENTER_X_MM),
        float(BOTTOM_NORTH_MARKER_GAP_MM),
        float(BOTTOM_NORTH_MARKER_INWARD_SHIFT_MM),
        float(NORTH_EXTRA_HOLE_CENTER_DISTANCE_MM),
        float(NORTH_EXTRA_HOLE_DIAMETER_MM),
        float(BOTTOM_ENGRAVING_DEPTH_MM),
        float(BOTTOM_ENGRAVING_EPS_MM),
        float(BOTTOM_INSERT_HEIGHT_MM),
        float(BOTTOM_INSERT_TOTAL_XY_REDUCTION_MM),
    )
    cutter = _BOTTOM_ENGRAVING_CUTTER_CACHE.get(cache_key)
    if cutter is None:
        if BOTTOM_NORTH_MARKER_TEXT != "N":
            raise ValueError('BOTTOM_NORTH_MARKER_TEXT must be "N".')

        # Preserve N font size, depth and orientation; move its cavity inward by 10 mm.
        text_size_reference = _make_text_cutter(
            BOTTOM_TEXT_SIZE_REFERENCE,
            0.0,
            target_height_mm=1.0e6,
            max_width_mm=BOTTOM_TEXT_SIZE_REFERENCE_WIDTH_MM,
            xy_scale=1.0,
        )
        common_text_height_mm = float(
            text_size_reference.bounds[1, 1] - text_size_reference.bounds[0, 1]
        )
        north_marker = _make_text_cutter(
            "N",
            0.0,
            target_height_mm=common_text_height_mm * BOTTOM_NORTH_MARKER_SCALE,
            max_width_mm=np.inf,
            xy_scale=1.0,
        )
        north_marker_center_y = (
            NORTH_EXTRA_HOLE_CENTER_DISTANCE_MM
            - 0.5 * NORTH_EXTRA_HOLE_DIAMETER_MM
            - BOTTOM_NORTH_MARKER_GAP_MM
            - BOTTOM_NORTH_MARKER_INWARD_SHIFT_MM
            - 0.5 * float(north_marker.bounds[1, 1] - north_marker.bounds[0, 1])
        )
        north_marker.apply_translation([
            BOTTOM_ENGRAVING_CENTER_X_MM, north_marker_center_y, 0.0
        ])
        cutter = north_marker
        _BOTTOM_ENGRAVING_CUTTER_CACHE[cache_key] = cutter.copy()
        _BOTTOM_ENGRAVING_INSERT_CACHE[cache_key] = (
            _make_compact_bottom_insert_from_cutters([north_marker])
        )
        print(f"N marker center Y: {north_marker_center_y:.2f} mm")
    else:
        cutter = cutter.copy()
        print("Reusing cached N-only engraving cutter and insert.")

    footprint_radius = float(np.max(np.linalg.norm(cutter.vertices[:, :2], axis=1)))
    flat_radius = min(INNER_BASE_DIAMETER_MM, INNER_BASE_DIAMETER2_MM) / 2.0 - INNER_BEVEL_MM
    if footprint_radius >= flat_radius:
        raise ValueError(
            f"N engraving reaches radius {footprint_radius:.2f} mm, "
            f"but common flat-bottom radius is only {flat_radius:.2f} mm."
        )

    body_m3 = _tm_to_manifold_direct(mesh, f"{variant_name} body")
    cutter_m3 = _tm_to_manifold_direct(cutter, f"{variant_name} N cutter")
    result_m3 = body_m3 - cutter_m3
    result = _manifold_to_tm_direct(result_m3)
    result = _clean_boolean_mesh(result, variant_name)
    print(
        f"N-only engraving OK: depth={BOTTOM_ENGRAVING_DEPTH_MM:.2f} mm; "
        f"insert XY clearance={BOTTOM_INSERT_TOTAL_XY_REDUCTION_MM:.2f} mm"
    )
    return result

def make_n_for_print():
    """Use the identical font/size/clearance as the previous N insert."""
    ref = _make_text_cutter(
        BOTTOM_TEXT_SIZE_REFERENCE, 0.0,
        target_height_mm=1.0e6,
        max_width_mm=BOTTOM_TEXT_SIZE_REFERENCE_WIDTH_MM,
        xy_scale=1.0,
    )
    actual_height = float(ref.bounds[1, 1] - ref.bounds[0, 1])
    cutter = _make_text_cutter(
        "N", 0.0,
        target_height_mm=actual_height * BOTTOM_NORTH_MARKER_SCALE,
        max_width_mm=np.inf,
        xy_scale=1.0,
    )
    letter = _make_bottom_insert_piece_from_cutter(cutter)
    mid_x = 0.5 * (letter.bounds[0, 0] + letter.bounds[1, 0])
    mid_y = 0.5 * (letter.bounds[0, 1] + letter.bounds[1, 1])
    letter.apply_translation([
        N_PRINT_CENTER_X_MM - mid_x,
        N_PRINT_CENTER_Y_MM - mid_y,
        -letter.bounds[0, 2],
    ])
    return letter


def load_and_position_target(path, label):
    # force='mesh' handles a Scene if the imported file contains multiple objects.
    target = trimesh.load(path, force="mesh", process=False)
    if not isinstance(target, trimesh.Trimesh) or len(target.faces) == 0:
        raise ValueError(f"{label}: downloaded STL has no usable triangular mesh")
    target = target.copy()
    center_x = 0.5 * float(target.bounds[0, 0] + target.bounds[1, 0])
    center_y = 0.5 * float(target.bounds[0, 1] + target.bounds[1, 1])
    dz = -float(target.bounds[0, 2]) if PLACE_TARGET_STL_ON_BED else 0.0
    target.apply_translation([
        -center_x if CENTER_TARGET_STL_ON_XY else 0.0,
        -center_y if CENTER_TARGET_STL_ON_XY else 0.0,
        dz,
    ])
    print(f"{label} target bounds after placement [mm]:\n{target.bounds}")
    print(f"{label} target watertight: {target.is_watertight}")
    return target


def prepare_annulus(mesh, rotation_degrees, label, extra_x_offset_mm=0.0):
    part = mesh.copy()
    # Rotate around the global Z axis BEFORE centering / grounding.
    if rotation_degrees:
        rot = trimesh.transformations.rotation_matrix(
            np.deg2rad(rotation_degrees), [0, 0, 1], [0, 0, 0]
        )
        part.apply_transform(rot)
    cx = 0.5 * float(part.bounds[0, 0] + part.bounds[1, 0])
    cy = 0.5 * float(part.bounds[0, 1] + part.bounds[1, 1])
    part.apply_translation([-cx + float(extra_x_offset_mm), -cy, -float(part.bounds[0, 2])])
    if not np.isclose(part.bounds[0, 2], 0.0, atol=1e-6):
        raise RuntimeError(f"{label}: bottom of annulus is not on Z=0")
    print(f"{label}: rotation +{rotation_degrees:.1f} deg CCW looking from +Z; extra X offset +{extra_x_offset_mm:.3f} mm")
    print(f"{label} bounds [mm]:\n{part.bounds}")
    return part


def write_compact_stl(target_path, annulus_source, rotation, output_path, label, annulus_extra_x_mm=0.0):
    target = load_and_position_target(target_path, label)
    ring = prepare_annulus(annulus_source, rotation, label, annulus_extra_x_mm)
    letter = make_n_for_print()
    # Concatenate triangle meshes into ONE STL, keeping three independent shells.
    # No boolean union: preserving exact dimensions is important for fitted pieces.
    assembly = trimesh.util.concatenate([target, ring, letter])
    if not np.all(np.isfinite(assembly.vertices)):
        raise RuntimeError(f"{label}: non-finite vertices in resulting STL")
    assembly.export(output_path)
    print(f"\n{label}: saved {output_path}")
    print(f"  N center XY: ({N_PRINT_CENTER_X_MM:.3f}, {N_PRINT_CENTER_Y_MM:.3f}) mm")
    print(f"  N bottom Z: {letter.bounds[0, 2]:.3f} mm")
    print(f"  Assembly bounds [mm]:\n{assembly.bounds}")
    print("  NOTE: 3 components in one STL; they are NOT boolean-fused.")
    return output_path

output1 = write_compact_stl(
