# OUTER ANNULUS ONLY: make a smoothed copy of the elevation raster.
# The original "reprojected" array remains untouched and is used by INNER.
def gaussian_filter_nan(data, sigma):
    if sigma <= 0:
        return data.copy()

    valid = np.isfinite(data)
    values = np.where(valid, data, 0.0)

    smooth_values = gaussian_filter(values, sigma=sigma, mode="nearest")
    smooth_weights = gaussian_filter(valid.astype(np.float64), sigma=sigma, mode="nearest")

    out = np.full_like(data, np.nan, dtype=np.float64)
    np.divide(
        smooth_values,
        smooth_weights,
        out=out,
        where=smooth_weights > 1e-12,
    )

    # Preserve the original NoData footprint.
    out[~valid] = np.nan
    return out

# ============================================================
# EXACT TANGENT-PLANE ORIENTATION: LOCAL EAST-NORTH-UP (ENU)
# ============================================================
# Goal for every final STL:
#   +X = EAST
#   +Y = NORTH
#   +Z = UP (away from Earth's center)
#
# IMPORTANT GEOMETRIC CONDITION:
# The circular INNER/ANNULUS cut is centered on XY=(0, 0).  The tangent plane
# through that center must therefore be the XY plane, and its normal (+/-Z)
# must pass EXACTLY through the center of the modeled Earth sphere.
#
# The previous version first oriented the raster using a spherical pixel
# centroid and then translated by the full 3-D mesh centroid.  The X/Y part of
# that translation moved the Earth center away from the Z axis.  Consequently
# the normal through the cut center no longer pointed exactly at the sphere
# center.
#
# Fix: use the actual terrain-surface centroid (the same physical center that
# the old code subsequently moved to the origin) to define the radial UP
# direction BEFORE any translation.  We then build an exact ENU frame at that
# direction.  After the rotation, this center lies on the Z axis, so only a
# Z translation is needed.  The Earth center therefore remains exactly on the
# Z axis below the cut center.


def exact_local_enu_frame_from_center(center_xyz, lon_deg_for_pole_fallback, pole_eps=1e-10):
    center_xyz = np.asarray(center_xyz, dtype=np.float64).reshape(3)
    center_norm = np.linalg.norm(center_xyz)
    if not np.isfinite(center_norm) or center_norm < 1e-12:
        raise RuntimeError("Could not determine a valid 3-D center for the terrain patch.")

    # Radial direction from Earth center to the chosen cut center.
    up = center_xyz / center_norm

    center_lat_rad = np.arcsin(np.clip(up[2], -1.0, 1.0))
    horizontal_norm = np.hypot(up[0], up[1])

    if horizontal_norm > pole_eps:
        center_lon_rad = np.arctan2(up[1], up[0])

        # EAST = increasing longitude.
        east = np.array([
            -np.sin(center_lon_rad),
             np.cos(center_lon_rad),
             0.0,
        ], dtype=np.float64)
        east /= np.linalg.norm(east)

        # NORTH = projection of Earth's north axis onto the tangent plane.
        earth_north_axis = np.array([0.0, 0.0, 1.0], dtype=np.float64)
        north = earth_north_axis - np.dot(earth_north_axis, up) * up
        north_norm = np.linalg.norm(north)
        if north_norm < 1e-12:
            raise RuntimeError("Numerical failure while constructing local NORTH axis.")
        north /= north_norm

        pole_fallback_used = False
    else:
        # Exact/near pole: geographic EAST/NORTH is singular.  Keep a
        # deterministic reference meridian derived from the raster longitudes.
        lon_rad = np.deg2rad(np.asarray(lon_deg_for_pole_fallback, dtype=np.float64))
        mean_sin_lon = np.mean(np.sin(lon_rad))
        mean_cos_lon = np.mean(np.cos(lon_rad))
        lon_resultant = np.hypot(mean_sin_lon, mean_cos_lon)

        if lon_resultant > 1e-12:
            center_lon_rad = np.arctan2(mean_sin_lon, mean_cos_lon)
        else:
            center_lon_rad = 0.0

        east = np.array([
            -np.sin(center_lon_rad),
             np.cos(center_lon_rad),
             0.0,
        ], dtype=np.float64)
        east /= np.linalg.norm(east)

        # Keep a right-handed frame: east x north = up.
        north = np.cross(up, east)
        north /= np.linalg.norm(north)
        pole_fallback_used = True

    # Final orthogonalization for numerical safety.
    east = east - np.dot(east, up) * up
    east /= np.linalg.norm(east)
    north = np.cross(up, east)
    north /= np.linalg.norm(north)

    # Preserve the true geographic NORTH sign away from the pole.
    if not pole_fallback_used:
        earth_north_axis = np.array([0.0, 0.0, 1.0], dtype=np.float64)
        north_reference = earth_north_axis - np.dot(earth_north_axis, up) * up
        if np.dot(north, north_reference) < 0:
            east = -east
            north = -north

    center_lon_deg = np.rad2deg(center_lon_rad)
    center_lat_deg = np.rad2deg(center_lat_rad)

    # Rows transform Earth-centered XYZ -> local ENU coordinates.
    rotation_3x3 = np.vstack([east, north, up])

    # Sanity checks.
    identity_error = np.max(np.abs(rotation_3x3 @ rotation_3x3.T - np.eye(3)))
    determinant = np.linalg.det(rotation_3x3)
    if identity_error > 1e-9 or determinant < 0.999999:
        raise RuntimeError(
            f"Invalid ENU rotation matrix: orthogonality error={identity_error}, "
            f"det={determinant}"
        )

    return (
        rotation_3x3,
        center_lon_deg,
        center_lat_deg,
        pole_fallback_used,
        center_norm,
    )

def interpolate_terrain_height(points_xy, z_shift=0.0):
    values = np.asarray(_terrain_linear(points_xy), dtype=np.float64).reshape(-1)
    missing = ~np.isfinite(values)
    if np.any(missing):
        _, nearest_idx = _terrain_nearest_tree.query(points_xy[missing], k=1)
        values[missing] = _base_z[nearest_idx]
    if z_shift != 0.0:
        values = values + z_shift
    return values


def interpolate_annulus_terrain_height(points_xy, z_shift=0.0):
    values = np.asarray(
        _annulus_terrain_linear(points_xy),
        dtype=np.float64,
    ).reshape(-1)
    missing = ~np.isfinite(values)
    if np.any(missing):
        _, nearest_idx = _annulus_terrain_nearest_tree.query(points_xy[missing], k=1)
        values[missing] = _annulus_base_z[nearest_idx]
    if z_shift != 0.0:
        values = values + z_shift
    return values


def interpolate_annulus2_terrain_height(points_xy, z_shift=0.0):
    values = np.asarray(
        _annulus2_terrain_linear(points_xy),
        dtype=np.float64,
    ).reshape(-1)
    missing = ~np.isfinite(values)
    if np.any(missing):
        _, nearest_idx = _annulus2_terrain_nearest_tree.query(points_xy[missing], k=1)
        values[missing] = _annulus2_base_z[nearest_idx]
    if z_shift != 0.0:
        values = values + z_shift
    return values


def terrain_at_top_height(base_mesh, top_height_mm):
    terrain = base_mesh.copy()
    max_z = terrain.vertices[:, 2].max()
    terrain.vertices[:, 2] -= (max_z - top_height_mm)
    return terrain


def validate_mesh(mesh, title):
    print("\n==============================")
    print(title)
    print("==============================")

    edges = mesh.edges_sorted
    _, counts = np.unique(edges, axis=0, return_counts=True)

# VECTORIZED LOOP CONNECTIVITY
# ============================================================
def connect_loop_faces(offset_a, offset_b, segments=SEGMENTS, reverse=False):
    """Return the same two triangles per loop segment as the original Python loops."""
    i = np.arange(segments, dtype=np.int64)
    n = (i + 1) % segments
    a0 = offset_a + i
    a1 = offset_a + n
    b0 = offset_b + i
    b1 = offset_b + n
    if reverse:
        return np.column_stack((
            np.column_stack((a0, b1, a1)),
            np.column_stack((a0, b0, b1)),
        )).reshape(-1, 3)
    return np.column_stack((
        np.column_stack((a0, a1, b1)),
        np.column_stack((a0, b1, b0)),
    )).reshape(-1, 3)


def bottom_fan_faces(center_id, bottom_offset, segments=SEGMENTS):
    i = np.arange(segments, dtype=np.int64)
    n = (i + 1) % segments
    return np.column_stack((
        np.full(segments, center_id, dtype=np.int64),
        bottom_offset + n,
        bottom_offset + i,
    ))

# OUTPUT 1: INNER CLOSED CYLINDER
# ============================================================

def build_inner_model(shell, hole_centers_xy, base_diameter_mm, variant_name="INNER"):
    """Build one INNER solid with one or more blind hemispherical-ended holes.

    hole_centers_xy is an iterable of (x, y) centers in final model coordinates:
      +X = EAST, -X = WEST, +Y = NORTH, -Y = SOUTH.

    base_diameter_mm selects which INNER base diameter this variant uses.
    """
    BASE_RADIUS_MM = float(base_diameter_mm) / 2.0
    BEVEL_MM = INNER_BEVEL_MM
    BEVEL_SEGMENTS = INNER_BEVEL_SEGMENTS
    TOP_BEVEL_MM = INNER_TOP_BEVEL_MM
    TOP_BEVEL_SEGMENTS = INNER_TOP_BEVEL_SEGMENTS
    HOLE_RADIUS_MM = INNER_BOTTOM_HOLE_DIAMETER_MM / 2.0
    HOLE_DEPTH_MM = INNER_BOTTOM_HOLE_DEPTH_MM
    HOLE_HEMISPHERE_SEGMENTS = INNER_BOTTOM_HOLE_HEMISPHERE_SEGMENTS

    hole_centers_xy = np.asarray(hole_centers_xy, dtype=np.float64).reshape(-1, 2)
    if len(hole_centers_xy) < 1:
        raise ValueError("At least one INNER bottom-hole center is required")

    theta = np.linspace(0, 2 * np.pi, SEGMENTS, endpoint=False)
    ct, st = np.cos(theta), np.sin(theta)

    # XY geometry is unchanged by the output-specific vertical shift.
    z_shift = shell.vertices[:, 2].max() - _BASE_MAX_Z

    if TOP_BEVEL_MM < 0 or TOP_BEVEL_MM >= BASE_RADIUS_MM:
        raise ValueError("Invalid INNER_TOP_BEVEL_MM")
    if TOP_BEVEL_SEGMENTS < 1:
        raise ValueError("INNER_TOP_BEVEL_SEGMENTS must be >= 1")

    top_inner_radius = BASE_RADIUS_MM - TOP_BEVEL_MM
    top_inner_xy = np.column_stack([top_inner_radius * ct, top_inner_radius * st])

    def interpolate_border_height(points_xy):
        return interpolate_terrain_height(points_xy, z_shift=z_shift)

    top_inner_z = interpolate_border_height(top_inner_xy)
    top_loop = np.column_stack([top_inner_xy, top_inner_z])

    r = np.linalg.norm(shell.vertices[:, :2], axis=1)
    terrain_points = shell.vertices[r <= top_inner_radius]

    points_xy = terrain_points[:, :2]
    tree = cKDTree(top_inner_xy)
    dist, _ = tree.query(points_xy, k=1)
    terrain_points = terrain_points[dist > 1e-6]

    cap_vertices = np.vstack([terrain_points, top_loop])
    top_boundary_offset = len(terrain_points)

    tri = Delaunay(cap_vertices[:, :2])
    top_faces = tri.simplices
    centers = np.mean(cap_vertices[top_faces][:, :, :2], axis=1)
    top_faces = top_faces[np.linalg.norm(centers, axis=1) <= top_inner_radius]

    top_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2) * k / TOP_BEVEL_SEGMENTS
            ring_r = (
                BASE_RADIUS_MM - TOP_BEVEL_MM
                + TOP_BEVEL_MM * np.sin(phi)
            )
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            ring_z = terrain_ring_z - bevel_drop
            top_bevel_rings.append(np.column_stack([ring_xy, ring_z]))

    if BEVEL_MM <= 0:
        bevel_rings = []
    else:
        if BEVEL_MM >= BASE_RADIUS_MM:
            raise ValueError("INNER_BEVEL_MM too large")
        if BEVEL_SEGMENTS < 1:
            raise ValueError("INNER_BEVEL_SEGMENTS must be >= 1")
        bevel_rings = []
        for k in range(BEVEL_SEGMENTS):
            phi = (np.pi / 2) * k / BEVEL_SEGMENTS
            ring_r = BASE_RADIUS_MM - BEVEL_MM + BEVEL_MM * np.cos(phi)
            ring_z = BEVEL_MM - BEVEL_MM * np.sin(phi)
            bevel_rings.append(np.column_stack([
                ring_r * ct,
                ring_r * st,
                np.full(SEGMENTS, ring_z),
            ]))

    bottom_radius = BASE_RADIUS_MM - max(BEVEL_MM, 0)
    bottom_loop = np.column_stack([
        bottom_radius * ct,
        bottom_radius * st,
        np.zeros(SEGMENTS),
    ])

    # ------------------------------------------------------------
    # BLIND HOLES FROM THE BOTTOM, EACH WITH A TRUE HEMISPHERICAL END
    # Diameter = 4 mm by default.
    # Straight cylindrical wall: Z=0 ... Z=HOLE_DEPTH_MM.
    # Hemisphere radius = hole radius, extending another radius upward.
    # Every cavity is closed, so the resulting mesh remains watertight.
    # ------------------------------------------------------------
    if HOLE_RADIUS_MM <= 0:
        raise ValueError("INNER_BOTTOM_HOLE_DIAMETER_MM must be > 0")
    if HOLE_DEPTH_MM <= 0:
        raise ValueError("INNER_BOTTOM_HOLE_DEPTH_MM must be > 0")
    if HOLE_HEMISPHERE_SEGMENTS < 2:
        raise ValueError("INNER_BOTTOM_HOLE_HEMISPHERE_SEGMENTS must be >= 2")

    # Every complete opening must remain inside the flat bottom and different
    # openings must not touch or overlap one another.
    center_distances = np.linalg.norm(hole_centers_xy, axis=1)
    outer_extents = center_distances + HOLE_RADIUS_MM
    if np.any(outer_extents >= bottom_radius):
        bad = int(np.argmax(outer_extents))
        raise ValueError(
            f"{variant_name}: bottom hole {bad + 1} intersects the lower bevel: "
            f"center distance + hole radius = {outer_extents[bad]:.3f} mm, "
            f"flat-bottom radius = {bottom_radius:.3f} mm"
        )

    if len(hole_centers_xy) > 1:
        delta = hole_centers_xy[:, None, :] - hole_centers_xy[None, :, :]
        pair_dist = np.linalg.norm(delta, axis=2)
        np.fill_diagonal(pair_dist, np.inf)
        min_pair = float(np.min(pair_dist))
        if min_pair <= 2.0 * HOLE_RADIUS_MM:
            raise ValueError(
                f"{variant_name}: bottom holes overlap or touch; "
                f"minimum center spacing={min_pair:.3f} mm"
            )

    hole_geometries = []
    deepest_z = HOLE_DEPTH_MM + HOLE_RADIUS_MM

    for hole_index, hole_center_xy in enumerate(hole_centers_xy):
        hole_xy = hole_center_xy + np.column_stack([
            HOLE_RADIUS_MM * ct,
            HOLE_RADIUS_MM * st,
        ])
        hole_bottom_loop = np.column_stack([hole_xy, np.zeros(SEGMENTS)])
        hole_end_loop = np.column_stack([
            hole_xy,
            np.full(SEGMENTS, HOLE_DEPTH_MM),
        ])

        hemisphere_rings = []
        for k in range(1, HOLE_HEMISPHERE_SEGMENTS):
            alpha = (np.pi / 2.0) * k / HOLE_HEMISPHERE_SEGMENTS
            ring_r = HOLE_RADIUS_MM * np.cos(alpha)
            ring_z = HOLE_DEPTH_MM + HOLE_RADIUS_MM * np.sin(alpha)
            ring_xy = hole_center_xy + np.column_stack([
                ring_r * ct,
                ring_r * st,
            ])
            hemisphere_rings.append(
                np.column_stack([ring_xy, np.full(SEGMENTS, ring_z)])
            )

        hemisphere_tip = np.array([[
            hole_center_xy[0],
            hole_center_xy[1],
            deepest_z,
        ]], dtype=np.float64)

        # Make sure the cavity cannot break through the terrain surface.
        check_xy = np.vstack([hole_xy, hole_center_xy.reshape(1, 2)])
        local_top_z = interpolate_border_height(check_xy)
        if deepest_z >= float(np.min(local_top_z)):
            raise ValueError(
                f"{variant_name}: bottom hole {hole_index + 1} "
                "(including hemispherical end) is too deep and would break "
                "through the terrain surface."
            )

        hole_geometries.append({
            "center": hole_center_xy,
            "bottom_loop": hole_bottom_loop,
            "end_loop": hole_end_loop,
            "hemisphere_rings": hemisphere_rings,
            "tip": hemisphere_tip,
        })

    parts = [cap_vertices]

    top_bevel_offsets = []
    for ring in top_bevel_rings:
        top_bevel_offsets.append(sum(len(x) for x in parts))
        parts.append(ring)

    bevel_offsets = []
    for ring in bevel_rings:
        bevel_offsets.append(sum(len(x) for x in parts))
        parts.append(ring)

    bottom_offset = sum(len(x) for x in parts)
    parts.append(bottom_loop)

    # Append each hole's opening loop, cylindrical end loop, hemispherical
    # latitude rings, and final tip. Keep their offsets for face generation.
    for hole in hole_geometries:
        hole["bottom_offset"] = sum(len(x) for x in parts)
        parts.append(hole["bottom_loop"])
        hole["end_offset"] = sum(len(x) for x in parts)
        parts.append(hole["end_loop"])

        hemisphere_offsets = []
        for ring in hole["hemisphere_rings"]:
            hemisphere_offsets.append(sum(len(x) for x in parts))
            parts.append(ring)
        hole["hemisphere_offsets"] = hemisphere_offsets

        hole["tip_id"] = sum(len(x) for x in parts)
        parts.append(hole["tip"])

    all_vertices = np.vstack(parts)
    face_parts = [np.asarray(top_faces, dtype=np.int64)]

    if top_bevel_offsets:
        previous_offset = top_boundary_offset
        for current_offset in top_bevel_offsets:
            face_parts.append(connect_loop_faces(previous_offset, current_offset))
            previous_offset = current_offset
        side_top_offset = top_bevel_offsets[-1]
    else:
        side_top_offset = top_boundary_offset

    side_bottom_offset = bevel_offsets[0] if bevel_offsets else bottom_offset
    face_parts.append(connect_loop_faces(side_top_offset, side_bottom_offset))

    for a_off, b_off in zip(bevel_offsets[:-1], bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off))

    if bevel_offsets:
        face_parts.append(connect_loop_faces(bevel_offsets[-1], bottom_offset))

    # ------------------------------------------------------------
    # TRIANGULATE THE FLAT BOTTOM WITH 1 OR MORE CIRCULAR HOLES
    # ------------------------------------------------------------
    # The old one-hole version could connect the outer loop directly to one
    # inner loop. With two independent holes that construction would overlap.
    # Shapely 2.1 constrained Delaunay triangulation respects the outer circle
    # and every circular hole boundary exactly, giving one clean planar face.
    outer_xy = bottom_loop[:, :2]
    hole_xy_rings = [h["bottom_loop"][:, :2] for h in hole_geometries]
    bottom_polygon = Polygon(outer_xy, holes=[ring for ring in hole_xy_rings])
    if (not bottom_polygon.is_valid) or bottom_polygon.is_empty:
        raise RuntimeError(f"{variant_name}: invalid bottom polygon with holes")

    bottom_triangles = shapely.constrained_delaunay_triangles(bottom_polygon)

    # Coordinate -> mesh vertex id. Constrained triangulation uses the supplied
    # boundary vertices; rounding only protects against tiny floating-point
    # representation differences when coordinates come back from GEOS.
    xy_to_vertex = {}
    for j, xy in enumerate(outer_xy):
        xy_to_vertex[(round(float(xy[0]), 10), round(float(xy[1]), 10))] = bottom_offset + j
    for hole in hole_geometries:
        off = hole["bottom_offset"]
        for j, xy in enumerate(hole["bottom_loop"][:, :2]):
            xy_to_vertex[(round(float(xy[0]), 10), round(float(xy[1]), 10))] = off + j

    bottom_faces = []
    for tri_geom in bottom_triangles.geoms:
        coords = np.asarray(tri_geom.exterior.coords, dtype=np.float64)[:3, :2]
        ids = []
        for xy in coords:
            key = (round(float(xy[0]), 10), round(float(xy[1]), 10))
            if key not in xy_to_vertex:
                raise RuntimeError(
                    f"{variant_name}: bottom triangulation introduced an unexpected vertex {xy}"
                )
            ids.append(xy_to_vertex[key])

        # Bottom outward normal must point toward -Z. A positive signed XY area
        # is +Z, so reverse such triangles.
        p0, p1, p2 = coords
        signed2 = (
            (p1[0] - p0[0]) * (p2[1] - p0[1])
            - (p1[1] - p0[1]) * (p2[0] - p0[0])
        )
        if signed2 > 0:
            ids = [ids[0], ids[2], ids[1]]
        bottom_faces.append(ids)

    face_parts.append(np.asarray(bottom_faces, dtype=np.int64))

    # Add the cylindrical wall + hemispherical termination for every hole.
    for hole in hole_geometries:
        face_parts.append(
            connect_loop_faces(
                hole["bottom_offset"],
                hole["end_offset"],
                reverse=True,
            )
        )

        previous_offset = hole["end_offset"]
        for current_offset in hole["hemisphere_offsets"]:
            face_parts.append(
                connect_loop_faces(previous_offset, current_offset, reverse=True)
            )
            previous_offset = current_offset

        last_ring_offset = (
            hole["hemisphere_offsets"][-1]
            if hole["hemisphere_offsets"]
            else hole["end_offset"]
        )
        i = np.arange(SEGMENTS, dtype=np.int64)
        n = (i + 1) % SEGMENTS
        face_parts.append(np.column_stack([
            last_ring_offset + i,
            np.full(SEGMENTS, hole["tip_id"], dtype=np.int64),
            last_ring_offset + n,
        ]))

    all_faces = np.vstack(face_parts)

    mesh = trimesh.Trimesh(
        vertices=all_vertices,
        faces=all_faces,
        process=False,
    )

    print(f"Cleaning {variant_name} mesh...")
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    mesh.merge_vertices()
    mesh.merge_vertices(digits_vertex=6)
    repair.fill_holes(mesh)
    mesh.fix_normals()
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    return mesh

def build_annulus_model(shell):
    OUTER_RADIUS_MM = ANNULUS_OUTER_DIAMETER_MM / 2.0
    INNER_RADIUS_MM = ANNULUS_INNER_DIAMETER_MM / 2.0
    BEVEL_MM = ANNULUS_BEVEL_MM
    BEVEL_SEGMENTS = ANNULUS_BEVEL_SEGMENTS
    TOP_BEVEL_MM = ANNULUS_TOP_BEVEL_MM
    TOP_BEVEL_SEGMENTS = ANNULUS_TOP_BEVEL_SEGMENTS

    if INNER_RADIUS_MM <= 0:
        raise ValueError("ANNULUS_INNER_DIAMETER_MM must be > 0")
    if INNER_RADIUS_MM >= OUTER_RADIUS_MM:
        raise ValueError("Annulus inner diameter must be smaller than outer diameter")
    if BEVEL_MM < 0 or TOP_BEVEL_MM < 0:
        raise ValueError("Annulus bevels must be >= 0")
    if BEVEL_SEGMENTS < 1 or TOP_BEVEL_SEGMENTS < 1:
        raise ValueError("Annulus bevel segment counts must be >= 1")
    if INNER_RADIUS_MM + BEVEL_MM >= OUTER_RADIUS_MM - BEVEL_MM:
        raise ValueError("ANNULUS_BEVEL_MM is too large for annulus width")
    if INNER_RADIUS_MM + TOP_BEVEL_MM >= OUTER_RADIUS_MM - TOP_BEVEL_MM:
        raise ValueError("ANNULUS_TOP_BEVEL_MM is too large for annulus width")

    theta = np.linspace(0.0, 2.0 * np.pi, SEGMENTS, endpoint=False)
    ct, st = np.cos(theta), np.sin(theta)

    # OUTER ANNULUS uses only the annulus-smoothed terrain/interpolator.
    z_shift = shell.vertices[:, 2].max() - _ANNULUS_BASE_MAX_Z

    def interpolate_border_height(points_xy):
        return interpolate_annulus_terrain_height(points_xy, z_shift=z_shift)

    top_outer_radius = OUTER_RADIUS_MM - TOP_BEVEL_MM
    top_inner_radius = INNER_RADIUS_MM + TOP_BEVEL_MM

    top_outer_xy = np.column_stack([top_outer_radius * ct, top_outer_radius * st])
    top_inner_xy = np.column_stack([top_inner_radius * ct, top_inner_radius * st])
    top_outer_z = interpolate_border_height(top_outer_xy)
    top_inner_z = interpolate_border_height(top_inner_xy)
    top_outer_loop = np.column_stack([top_outer_xy, top_outer_z])
    top_inner_loop = np.column_stack([top_inner_xy, top_inner_z])

    r = np.linalg.norm(shell.vertices[:, :2], axis=1)
    terrain_points = shell.vertices[(r >= top_inner_radius) & (r <= top_outer_radius)]

    if len(terrain_points) > 0:
        points_xy = terrain_points[:, :2]
        outer_dist, _ = cKDTree(top_outer_xy).query(points_xy, k=1)
        inner_dist, _ = cKDTree(top_inner_xy).query(points_xy, k=1)
        terrain_points = terrain_points[(outer_dist > 1e-6) & (inner_dist > 1e-6)]

    cap_vertices = np.vstack([terrain_points, top_outer_loop, top_inner_loop])
    top_outer_boundary_offset = len(terrain_points)
    top_inner_boundary_offset = top_outer_boundary_offset + SEGMENTS

    tri = Delaunay(cap_vertices[:, :2])
    top_faces = tri.simplices
    centers = np.mean(cap_vertices[top_faces][:, :, :2], axis=1)
    center_r = np.linalg.norm(centers, axis=1)
    top_faces = top_faces[(center_r <= top_outer_radius) & (center_r >= top_inner_radius)]

    top_outer_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2.0) * k / TOP_BEVEL_SEGMENTS
            ring_r = OUTER_RADIUS_MM - TOP_BEVEL_MM + TOP_BEVEL_MM * np.sin(phi)
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            top_outer_bevel_rings.append(
                np.column_stack([ring_xy, terrain_ring_z - bevel_drop])
            )

    top_inner_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2.0) * k / TOP_BEVEL_SEGMENTS
            ring_r = INNER_RADIUS_MM + TOP_BEVEL_MM - TOP_BEVEL_MM * np.sin(phi)
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            top_inner_bevel_rings.append(
                np.column_stack([ring_xy, terrain_ring_z - bevel_drop])
            )

    bottom_outer_bevel_rings = []
    if BEVEL_MM > 0:
        for k in range(BEVEL_SEGMENTS):
            phi = (np.pi / 2.0) * k / BEVEL_SEGMENTS
            ring_r = OUTER_RADIUS_MM - BEVEL_MM + BEVEL_MM * np.cos(phi)
            ring_z = BEVEL_MM - BEVEL_MM * np.sin(phi)
            bottom_outer_bevel_rings.append(np.column_stack([
                ring_r * ct,
                ring_r * st,
                np.full(SEGMENTS, ring_z),
            ]))

    bottom_inner_bevel_rings = []
    if BEVEL_MM > 0:
        for k in range(BEVEL_SEGMENTS):
            phi = (np.pi / 2.0) * k / BEVEL_SEGMENTS
            ring_r = INNER_RADIUS_MM + BEVEL_MM - BEVEL_MM * np.cos(phi)
            ring_z = BEVEL_MM - BEVEL_MM * np.sin(phi)
            bottom_inner_bevel_rings.append(np.column_stack([
                ring_r * ct,
                ring_r * st,
                np.full(SEGMENTS, ring_z),
            ]))

    bottom_outer_radius = OUTER_RADIUS_MM - BEVEL_MM
    bottom_inner_radius = INNER_RADIUS_MM + BEVEL_MM
    bottom_outer_loop = np.column_stack([
        bottom_outer_radius * ct,
        bottom_outer_radius * st,
        np.zeros(SEGMENTS),
    ])
    bottom_inner_loop = np.column_stack([
        bottom_inner_radius * ct,
        bottom_inner_radius * st,
        np.zeros(SEGMENTS),
    ])

    parts = [cap_vertices]

    def append_rings(rings):
        offsets = []
        for ring in rings:
            offsets.append(sum(len(x) for x in parts))
            parts.append(ring)
        return offsets

    top_outer_bevel_offsets = append_rings(top_outer_bevel_rings)
    top_inner_bevel_offsets = append_rings(top_inner_bevel_rings)
    bottom_outer_bevel_offsets = append_rings(bottom_outer_bevel_rings)
    bottom_inner_bevel_offsets = append_rings(bottom_inner_bevel_rings)

    bottom_outer_offset = sum(len(x) for x in parts)
    parts.append(bottom_outer_loop)
    bottom_inner_offset = sum(len(x) for x in parts)
    parts.append(bottom_inner_loop)

    all_vertices = np.vstack(parts)
    face_parts = [np.asarray(top_faces, dtype=np.int64)]

    previous_offset = top_outer_boundary_offset
    for current_offset in top_outer_bevel_offsets:
        face_parts.append(connect_loop_faces(previous_offset, current_offset, reverse=False))
        previous_offset = current_offset
    outer_wall_top_offset = previous_offset

    previous_offset = top_inner_boundary_offset
    for current_offset in top_inner_bevel_offsets:
        face_parts.append(connect_loop_faces(previous_offset, current_offset, reverse=True))
        previous_offset = current_offset
    inner_wall_top_offset = previous_offset

    outer_wall_bottom_offset = (
        bottom_outer_bevel_offsets[0]
        if bottom_outer_bevel_offsets else bottom_outer_offset
    )
    inner_wall_bottom_offset = (
        bottom_inner_bevel_offsets[0]
        if bottom_inner_bevel_offsets else bottom_inner_offset
    )

    face_parts.append(connect_loop_faces(outer_wall_top_offset, outer_wall_bottom_offset, reverse=False))
    face_parts.append(connect_loop_faces(inner_wall_top_offset, inner_wall_bottom_offset, reverse=True))

    for a_off, b_off in zip(bottom_outer_bevel_offsets[:-1], bottom_outer_bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off, reverse=False))
    if bottom_outer_bevel_offsets:
        face_parts.append(connect_loop_faces(bottom_outer_bevel_offsets[-1], bottom_outer_offset, reverse=False))

    for a_off, b_off in zip(bottom_inner_bevel_offsets[:-1], bottom_inner_bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off, reverse=True))
    if bottom_inner_bevel_offsets:
        face_parts.append(connect_loop_faces(bottom_inner_bevel_offsets[-1], bottom_inner_offset, reverse=True))

    face_parts.append(connect_loop_faces(bottom_outer_offset, bottom_inner_offset, reverse=True))
    all_faces = np.vstack(face_parts)

    mesh = trimesh.Trimesh(
        vertices=all_vertices,
        faces=all_faces,
        process=False,
    )

    print("Cleaning annulus mesh...")
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    mesh.merge_vertices()
    mesh.merge_vertices(digits_vertex=6)
    repair.fill_holes(mesh)
    mesh.fix_normals()
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    return mesh

# OUTPUT 3: OUTER ANNULUS 2 - NO BOTTOM INNER BEVEL
# ============================================================

def build_annulus2_model(shell):
    OUTER_RADIUS_MM = ANNULUS2_OUTER_DIAMETER_MM / 2.0
    INNER_RADIUS_MM = ANNULUS2_INNER_DIAMETER_MM / 2.0
    BEVEL_MM = ANNULUS2_OUTER_BOTTOM_BEVEL_MM
    BEVEL_SEGMENTS = ANNULUS2_OUTER_BOTTOM_BEVEL_SEGMENTS
    TOP_BEVEL_MM = ANNULUS2_TOP_BEVEL_MM
    TOP_BEVEL_SEGMENTS = ANNULUS2_TOP_BEVEL_SEGMENTS

    if INNER_RADIUS_MM <= 0:
        raise ValueError("ANNULUS2_INNER_DIAMETER_MM must be > 0")
    if INNER_RADIUS_MM >= OUTER_RADIUS_MM:
        raise ValueError("Annulus 2 inner diameter must be smaller than outer diameter")
    if BEVEL_MM < 0 or TOP_BEVEL_MM < 0:
        raise ValueError("Annulus 2 bevels must be >= 0")
    if BEVEL_SEGMENTS < 1 or TOP_BEVEL_SEGMENTS < 1:
        raise ValueError("Annulus 2 bevel segment counts must be >= 1")
    if INNER_RADIUS_MM >= OUTER_RADIUS_MM - BEVEL_MM:
        raise ValueError("ANNULUS2_OUTER_BOTTOM_BEVEL_MM is too large for annulus width")
    if INNER_RADIUS_MM + TOP_BEVEL_MM >= OUTER_RADIUS_MM - TOP_BEVEL_MM:
        raise ValueError("ANNULUS2_TOP_BEVEL_MM is too large for annulus width")

    theta = np.linspace(0.0, 2.0 * np.pi, SEGMENTS, endpoint=False)
    ct, st = np.cos(theta), np.sin(theta)

    # OUTER ANNULUS uses only the annulus-smoothed terrain/interpolator.
    z_shift = shell.vertices[:, 2].max() - _ANNULUS2_BASE_MAX_Z

    def interpolate_border_height(points_xy):
        return interpolate_annulus2_terrain_height(points_xy, z_shift=z_shift)

    top_outer_radius = OUTER_RADIUS_MM - TOP_BEVEL_MM
    top_inner_radius = INNER_RADIUS_MM + TOP_BEVEL_MM

    top_outer_xy = np.column_stack([top_outer_radius * ct, top_outer_radius * st])
    top_inner_xy = np.column_stack([top_inner_radius * ct, top_inner_radius * st])
    top_outer_z = interpolate_border_height(top_outer_xy)
    top_inner_z = interpolate_border_height(top_inner_xy)
    top_outer_loop = np.column_stack([top_outer_xy, top_outer_z])
    top_inner_loop = np.column_stack([top_inner_xy, top_inner_z])

    r = np.linalg.norm(shell.vertices[:, :2], axis=1)
    terrain_points = shell.vertices[(r >= top_inner_radius) & (r <= top_outer_radius)]

    if len(terrain_points) > 0:
        points_xy = terrain_points[:, :2]
        outer_dist, _ = cKDTree(top_outer_xy).query(points_xy, k=1)
        inner_dist, _ = cKDTree(top_inner_xy).query(points_xy, k=1)
        terrain_points = terrain_points[(outer_dist > 1e-6) & (inner_dist > 1e-6)]

    cap_vertices = np.vstack([terrain_points, top_outer_loop, top_inner_loop])
    top_outer_boundary_offset = len(terrain_points)
    top_inner_boundary_offset = top_outer_boundary_offset + SEGMENTS

    tri = Delaunay(cap_vertices[:, :2])
    top_faces = tri.simplices
    centers = np.mean(cap_vertices[top_faces][:, :, :2], axis=1)
    center_r = np.linalg.norm(centers, axis=1)
    top_faces = top_faces[(center_r <= top_outer_radius) & (center_r >= top_inner_radius)]

    top_outer_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2.0) * k / TOP_BEVEL_SEGMENTS
            ring_r = OUTER_RADIUS_MM - TOP_BEVEL_MM + TOP_BEVEL_MM * np.sin(phi)
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            top_outer_bevel_rings.append(
                np.column_stack([ring_xy, terrain_ring_z - bevel_drop])
            )

    top_inner_bevel_rings = []
    if TOP_BEVEL_MM > 0:
        for k in range(1, TOP_BEVEL_SEGMENTS + 1):
            phi = (np.pi / 2.0) * k / TOP_BEVEL_SEGMENTS
            ring_r = INNER_RADIUS_MM + TOP_BEVEL_MM - TOP_BEVEL_MM * np.sin(phi)
            ring_xy = np.column_stack([ring_r * ct, ring_r * st])
            terrain_ring_z = interpolate_border_height(ring_xy)
            bevel_drop = TOP_BEVEL_MM * (1.0 - np.cos(phi))
            top_inner_bevel_rings.append(
                np.column_stack([ring_xy, terrain_ring_z - bevel_drop])
            )

    bottom_outer_bevel_rings = []
    if BEVEL_MM > 0:
        for k in range(BEVEL_SEGMENTS):
            phi = (np.pi / 2.0) * k / BEVEL_SEGMENTS
            ring_r = OUTER_RADIUS_MM - BEVEL_MM + BEVEL_MM * np.cos(phi)
            ring_z = BEVEL_MM - BEVEL_MM * np.sin(phi)
            bottom_outer_bevel_rings.append(np.column_stack([
                ring_r * ct,
                ring_r * st,
                np.full(SEGMENTS, ring_z),
            ]))

    # ANNULUS 2: intentionally NO bottom inner bevel.
    # The inner cylindrical wall continues straight down to Z=0 at INNER_RADIUS_MM.
    bottom_inner_bevel_rings = []

    bottom_outer_radius = OUTER_RADIUS_MM - BEVEL_MM
    bottom_inner_radius = INNER_RADIUS_MM
    bottom_outer_loop = np.column_stack([
        bottom_outer_radius * ct,
        bottom_outer_radius * st,
        np.zeros(SEGMENTS),
    ])
    bottom_inner_loop = np.column_stack([
        bottom_inner_radius * ct,
        bottom_inner_radius * st,
        np.zeros(SEGMENTS),
    ])

    parts = [cap_vertices]

    def append_rings(rings):
        offsets = []
        for ring in rings:
            offsets.append(sum(len(x) for x in parts))
            parts.append(ring)
        return offsets

    top_outer_bevel_offsets = append_rings(top_outer_bevel_rings)
    top_inner_bevel_offsets = append_rings(top_inner_bevel_rings)
    bottom_outer_bevel_offsets = append_rings(bottom_outer_bevel_rings)
    bottom_inner_bevel_offsets = append_rings(bottom_inner_bevel_rings)

    bottom_outer_offset = sum(len(x) for x in parts)
    parts.append(bottom_outer_loop)
    bottom_inner_offset = sum(len(x) for x in parts)
    parts.append(bottom_inner_loop)

    all_vertices = np.vstack(parts)
    face_parts = [np.asarray(top_faces, dtype=np.int64)]

    previous_offset = top_outer_boundary_offset
    for current_offset in top_outer_bevel_offsets:
        face_parts.append(connect_loop_faces(previous_offset, current_offset, reverse=False))
        previous_offset = current_offset
    outer_wall_top_offset = previous_offset

    previous_offset = top_inner_boundary_offset
    for current_offset in top_inner_bevel_offsets:
        face_parts.append(connect_loop_faces(previous_offset, current_offset, reverse=True))
        previous_offset = current_offset
    inner_wall_top_offset = previous_offset

    outer_wall_bottom_offset = (
        bottom_outer_bevel_offsets[0]
        if bottom_outer_bevel_offsets else bottom_outer_offset
    )
    inner_wall_bottom_offset = (
        bottom_inner_bevel_offsets[0]
        if bottom_inner_bevel_offsets else bottom_inner_offset
    )

    face_parts.append(connect_loop_faces(outer_wall_top_offset, outer_wall_bottom_offset, reverse=False))
    face_parts.append(connect_loop_faces(inner_wall_top_offset, inner_wall_bottom_offset, reverse=True))

    for a_off, b_off in zip(bottom_outer_bevel_offsets[:-1], bottom_outer_bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off, reverse=False))
    if bottom_outer_bevel_offsets:
        face_parts.append(connect_loop_faces(bottom_outer_bevel_offsets[-1], bottom_outer_offset, reverse=False))

    for a_off, b_off in zip(bottom_inner_bevel_offsets[:-1], bottom_inner_bevel_offsets[1:]):
        face_parts.append(connect_loop_faces(a_off, b_off, reverse=True))
    if bottom_inner_bevel_offsets:
        face_parts.append(connect_loop_faces(bottom_inner_bevel_offsets[-1], bottom_inner_offset, reverse=True))

    face_parts.append(connect_loop_faces(bottom_outer_offset, bottom_inner_offset, reverse=True))
    all_faces = np.vstack(face_parts)

    mesh = trimesh.Trimesh(
        vertices=all_vertices,
        faces=all_faces,
        process=False,
    )

    print("Cleaning annulus 2 mesh...")
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    mesh.merge_vertices()
    mesh.merge_vertices(digits_vertex=6)
    repair.fill_holes(mesh)
    mesh.fix_normals()
    mesh.update_faces(mesh.unique_faces())
    mesh.remove_unreferenced_vertices()
    return mesh
